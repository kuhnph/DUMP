#!/usr/bin/env python2

import sys
import os

import simwbPath
import simwbRT
import simwbConstants


# =============================================================================
# RTDB VARIABLE
# =============================================================================

TIME_RTDB = "SW_T_Simulation"


# =============================================================================
# CONNECT TO SIMWB
# =============================================================================

simwbRT.mapRTDB()

simwbRT.initLog(
    os.path.basename(sys.argv[1])
)

simwbRT.initSched(
    sys.argv[1]
)


# =============================================================================
# GET SIMULATION FRAME INFORMATION
# =============================================================================

frame_rate = simwbRT.getFrameRate()

if frame_rate <= 0:
    simwbRT.logMessage(
        "Invalid SimWB frame rate: %s"
        % str(frame_rate)
    )

    simwbRT.initFailed()
    sys.exit(1)


dt = 1.0 / float(frame_rate)


simwbRT.logMessage(
    "Time model initialized: frame rate = %.3f Hz, dt = %.9f s"
    % (
        frame_rate,
        dt
    )
)


# =============================================================================
# SCHEDULER INITIALIZATION CYCLES
# =============================================================================

try:

    for i in range(simwbConstants.SCHED_INITCYCLES):

        simwbRT.waitNextFrame()


    # =========================================================================
    # REAL-TIME LOOP
    # =========================================================================

    while True:

        # Wait for the next SimWB frame
        simwbRT.waitNextFrame()


        # ---------------------------------------------------------------------
        # Get current frame number
        # ---------------------------------------------------------------------

        frame_count = simwbRT.getFrameCount()


        # ---------------------------------------------------------------------
        # Convert frame count to simulation time
        #
        # t = frame_number * timestep
        # ---------------------------------------------------------------------

        t_simulation = float(frame_count) * dt


        # ---------------------------------------------------------------------
        # Write simulation time into the RTDB
        # ---------------------------------------------------------------------

        simwbRT.setCVTValue(
            TIME_RTDB,
            [t_simulation]
        )


except KeyboardInterrupt:

    simwbRT.logMessage(
        "Simulation time model shutting down."
    )

    sys.exit(0)
