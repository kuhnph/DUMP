#define POINTGROUP_MASK  0xf
enum pointGroup {
     SimWB_ANALOG  = 1,
     SimWB_BOOLEAN = 2,
     SimWB_MBIT    = 3,
     SimWB_STRING  = 4,
     SimWB_PARAMETER= 8,
     SimWB_SHARED_PARAMETER = 10,
     SimWB_MDLSIGNAL= 12
};

#define POINTDIRECTION_MASK 0xf0
enum pointDirection {
    SimWB_DIR_INPUT = 0x10,
    SimWB_DIR_OUTPUT= 0x20,
    SimWB_DIR_INOUT = 0x30
};
#define POINT_MASK (POINTGROUP_MASK | POINTDIRECTION_MASK) 

#define POINTDIRECTIONRT_MASK 0xf000
enum pointDirectionRT {
    SimWB_DIR_RTINPUT = 0x1000,
    SimWB_DIR_RTOUTPUT= 0x2000
};

#define RTDB_ISINPUT(pType) ((pType & POINTDIRECTION_MASK) & SimWB_DIR_INPUT)
#define RTDB_ISOUTPUT(pType) ((pType & POINTDIRECTION_MASK) & SimWB_DIR_OUTPUT)
#define RTDB_ISDIGITAL(pType) ((pType & POINTGROUP_MASK) == SimWB_BOOLEAN )
#define RTDB_ISANALOG(pType) ((pType & POINTGROUP_MASK) == SimWB_ANALOG )
#define RTDB_ISSTRING(pType) ((pType & POINTGROUP_MASK) == SimWB_STRING )

#define RTDB_ISRTINPUT(pType) (pType  & SimWB_DIR_RTINPUT)
#define RTDB_ISRTOUTPUT(pType) (pType & SimWB_DIR_RTOUTPUT)

/**
 * \ingroup ccurRTDBUtils
 * \ingroup RTDBAccess
 * Object to hold the engineering unit value of a RTDB item independent of its data type.
 * This is used in call to getCVTValue and getAlternateValue.
 */
typedef struct ItemValue {
    int stringLength;
    int numElements;            /**< The number of elements for array type items. */
    int numMallocedElements;    /**< size of the buffer for getCVT values if it has been preallocated by the caller. for getCVTValue, etc... This can be less than the number of elements.*/
    int isRTDB ;                /**< Indicates whether the string comes from RTDB (true) or local storage (false). */
    union {
        union {
            long long ll __attribute__ ((aligned(8))) ;           /**< Value of the item if its cvt type is long long (64 bit integer).*/
            double d __attribute__ ((aligned(8))) ;               /**< Value of the item if its cvt type is double */
            float f;                /**< Value of the item if its cvt type is float */
            short s;                /**< Value of the item if its cvt type is short */
            unsigned short us;      /**< Value of the item if its cvt type is unsigned short */
            char           c;       /**< Value of the item if its cvt type is char */
            unsigned char  uc;      /**< Value of the item if its cvt type is unsigned char */
            int            i;       /**< Value of the item if its cvt type is int */
            unsigned int   ui;      /**< Value of the item if its cvt type is unsigned int */
            short          stringIndex; /**< Index of the string value in the string table */
            StringVariableValue *stringValue __attribute__ ((aligned(8))) ; /**< For temporary storage (Not referring to RTDB values. */
        } Scalar __attribute__ ((aligned(8))) ;
        //RTDBArrayValue *Array;
        char *array_c __attribute__ ((aligned(8))) ;
        unsigned char *array_uc __attribute__ ((aligned(8))) ;
        short *array_s __attribute__ ((aligned(8))) ;
        unsigned short *array_us __attribute__ ((aligned(8))) ;
        int *array_i __attribute__ ((aligned(8))) ;
        unsigned int *array_ui __attribute__ ((aligned(8))) ;
        float *array_f __attribute__ ((aligned(8))) ;
        double *array_d __attribute__ ((aligned(8))) ;
        long long *array_ll __attribute__ ((aligned(8))) ;
        void *array_void __attribute__ ((aligned(8))) ;
    } Val;
    int dimensions[MAX_DIMENSIONS] __attribute__ ((aligned(8))) ;
} ItemValue;

#define EU_RAWTOPHYSICAL 0x80
/**
 * Type of entry in the conversion table. \sa EUConversion .
 * \ingroup ccurRTDBUtils
 * \ingroup RTDBAccess
 */
enum EUConversionType {
    EU_RATFUNC = 1 , /**< Rational function .*/
    EU_TABINTERP = 2 , /**< Table with interpolation. */
    EU_TABNOINTERP = 3   /**< Table with no interpolation. */
};
#define EU_MAXTABLEENTRIES  50
#define EU_MAXPOLYNOMIAL    3

#define EU_LINEARCONVERSION 0x1
#define EU_1OVERXCONVERSION 0x2

#define EU_ERRORNOINVERSION 0x1
/**
 * \ingroup ccurRTDBUtils
 * \ingroup RTDBAccess
 * Engineering conversion record held in shared memory. 
 * This is indexed via a separate hash table pointed to in the EUFactors meta data.
 * This defines a conversion from physical (Engineering units) to raw via either a
 * rational polynomial function as raw = (a*X^2 + b*X + c)/(d*X^2 + e*X + F) or <BR>
 * an interpolation table defined in rangesX and rangesY.<BR>
 * When the polynomial is different from strictly linear , i.e. A*X + B, this function cannot
 * be inverted and therefore cannot be used to convert from physical to raw units.
 */
typedef struct EUConversion {
    int   hashIndex;                    /**< Index # in hash table for this item */
    char  name[MAX_EUCONVERSIONNAME];   /**< Name of this conversion. */
    int   type;                         /**< Type of conversion \sa EUConversionType. By default, conversion
                                         is physical to raw. raw to physical to raw defined by oring with EU_RAWTOPHYSICAL */
    int   flags;                        /**< Whether the polynomial is linear and can be inverted.*/
    double numerators[EU_MAXPOLYNOMIAL] __attribute__ ((aligned(8))) ;    /**< a,b,c coefficients. */       
    double denominators[EU_MAXPOLYNOMIAL] __attribute__ ((aligned(8))) ;   /**< d,e,f coefficients. */
    int    nRanges;                     /**< Number of entries in interpolation table. */
    double rangesX[EU_MAXTABLEENTRIES] __attribute__ ((aligned(8))) ;
    double rangesY[EU_MAXTABLEENTRIES] __attribute__ ((aligned(8))) ;
    double rangesInvertedX[EU_MAXTABLEENTRIES] __attribute__ ((aligned(8))) ;
    double rangesInvertedY[EU_MAXTABLEENTRIES] __attribute__ ((aligned(8))) ;
    double directA __attribute__ ((aligned(8))) ;                  /**< For linear conversion only. Those are calculated a load time. Raw = directA * Physical_value  + directB .*/
    double directB __attribute__ ((aligned(8))) ;
    double inverseA __attribute__ ((aligned(8))) ;                  /**< For linear conversion only.  Pysical_value = inverseA * Raw  + inverseB .*/
    double inverseB __attribute__ ((aligned(8))) ;
    char   description[MAX_DESCRIPTIONLEN];   /**< Conversion description */
    char   engUnits[MAX_EUNITSLEN];           /**< Engineering unit descriptor for physical units. */
} EUConversion;

/**
 * \ingroup ccurRTDBUtils
 * \ingroup RTDBAccess
 * Meta flags for an RTDB item. Describe specific properties of the  RTDB item such as : item has EU conversion, is logged in the data logger,etc.
 */
enum RTDB_MetaFlags {
    METAFL_HASEUCONVERSION  =1,     /**< Use EU conversion factors A & B to convert raw value. If not set there is no conversion and the raw value is stored as is in the RTDB */
    METAFL_HASDEFAULT       =2,     /**< Item has a default value associated with it. The RTDB item is initialized to this value when the RTDB is loaded and the item value
                                         has not been specified in the initial condition set.*/
    METAFL_HASEULIMITS      =4,     /**< Check EU limits for valid value when placing in RTDB. This only applies to RTDB item mapped to hardware device.*/
    METAFL_ARCHIVE          =8,     /**< Log the value of this item to the historical files at run time. This puts the point in the list of point that can be logged. 
                                         If this is on, METFAFL_ARCHIVEENABLE is turned on at initialization. At run time, the user can toggle the METAFL_ARCHIVENABLE flag 
                                         to enable/disable the logging. */
    METAFL_ARCHIVEENABLE    =0x10,  /**< Dynamic logging flag. When on, the item is logged when off item is no longer logged. This flag can be toggled at run time. */
    METAFL_INVERTLOGICAL    =0x20,  /**< Set when boolean raw data must be inverted before storing in the CVT. */
    METAFL_SIGNEDEUVAL      =0x40,  /**< Set  the EU value is a signed number, i.e. the most significant bit of the hardware raw value is the sign bit. */
    METAFL_CLAMP2EULIMIT    =0x80,  /**< If set and HASEULIMITS also set, the input and output values EU values are clamped to the max/min defined by the EU limits. */
    METAFL_NOALTVALUE       =0x100, /**< If set , the item does not have an alternate value. */
    METAFL_PLAYBACK         =0x200, /**< If set , we're in playback mode and the item has data coming from the data logger . */
    METAFL_CIRCULARLOGGER   =0x400, /**< If set , this item goes into the circular data logger file . */
    METAFL_READONLY         =0x800, /**< Read only item, value cannot be set from the GUI.*/
    METAFL_EUCONVRECORD     =0x1000, /**< Use EU conversion in table instead  of A,B factors . */
    METAFL_ASYNCENTRY       =0x2000, /**< Set by asyncio input thread when new value is set so that I/O tasks can output even when value has not changed.*/
    METAFL_COLMAJOR         =0x4000, /**< Col major for MATLAB type arrays. */    
    METAFL_SPARE            =0x8000, /**< Last spare bit. */
    METAFL_NOSTRUCT         =0x10000  /**< Don't include this item in the C CVTTABLE struct. */
};
