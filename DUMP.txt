
%$Insert-OUO-Block
function [parameterList, structNames] = getHg9900ParameterList(options)
% function [parameterList, structNames] = getHg9900ParameterList(options)
%
% Gets the list of parameters to properly read the raw IMU data.
%
% NAME/VALUE ARGUMENTS
% --------------------
% RecordingName
% VirtualChannelName
% PacketName
% ParamPrefix
% GyroCount
% ParamMsgNumber
% ParamTimestamp
% ParamAccelTag
% ParamAccelWord
% ParamGyroTagTemplate
% ParamGyroWordTemplate
% ParamGyroStatusTemplate
% ParamGyroThetaLsbTemplate
% ParamGyroThetaMsbTemplate
% ParamGyroChecksumTemplate

arguments
    % Name of the TM recording.
    options.RecordingName char {mustBeTextScalar} = '';

    % Name of the CCSDS Virtual Channel.
    options.VirtualChannelName char {mustBeTextScalar} = 'Navigation';

    % Name of the CCSDS Space Packet
    options.PacketName char ...
        {mustBeTextScalar} = ['...MinAtr...IMU...HG9900...Space_Packet' ...
        '--HG9900_Raw_Data'];

    % A prefix added to the ParameterName when loading from the packet.
    options.ParamPrefix char {mustBeTextScalar} = '';

    % The number of gyrometer channels available within the IMU.
    options.GyroCount(1,1) double = 3;

    % The parameter name of the message number.
    options.ParamMsgNumber char ...
        {mustBeTextScalar} = ['/MinAtr/IMU/HG9900/' ...
        'HG9900_Raw_Data_Frame--Msg_Number'];

    % The parameter name of the time stamp.
    options.ParamTimestamp char ...
        {mustBeTextScalar} = ['/MinAtr/IMU/HG9900/' ...
        'HG9900_Raw_Data_Frame--Time_Of_Validity'];

    % The parameter name of the accelerometer tag.
    options.ParamAccelTag char ...
        {mustBeTextScalar} = ['/MinAtr/IMU/HG9900/' ...
        'HG9900_Raw_Data_Frame--Raptor_Counts--accel_mux_tag'];

    % The parameter name of the accelerometer word.
    options.ParamAccelWord char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
        'HG9900_Raw_Data_Frame--Raptor_Counts--accel_mux_word'];

    % The sprintf template for the parameter name of the gyrometer
    % tag.
    options.ParamGyroTagTemplate char ...
        {mustBeTextScalar} = ['/MinAtr/IMU/HG9900/' ...
        'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%d--mux_tag'];

    % The sprintf template for the parameter name of the gyrometer
    % word.
    options.ParamGyroWordTemplate ...
        char {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
        'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%d--mux_word'];

    % The sprintf template for the parameter name of the gyrometer
    % status.
    options.ParamGyroStatusTemplate ...
        char {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
        'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%d--status'];

    % The sprintf template for the parameter name of the gyrometer
    % theta lsb.
    options.ParamGyroThetaLsbTemplate ...
        char {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
        'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%d--lsb'];

    % The sprintf template for the parameter name of the gyrometer
    % theta msb.
    options.ParamGyroThetaMsbTemplate ...
        char {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
        'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%d--msb'];

    % The sprintf template for the parameter name of the gyrometer
    % theta msb.
    options.ParamGyroChecksumTemplate ...
        char {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
        'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%d--checksum'];
end

import MinAtrTmAnalysis.Framework.Data.ParameterSpec

accelNames = {
    options.ParamMsgNumber;
    options.ParamTimestamp;
    options.ParamAccelTag;
    options.ParamAccelWord;
    };

accelStructNames = {
    'HG9900_Raw_Data_Frame_Msg_Number';
    'HG9900_Raw_Data_Frame_Timestamp';
    'HG9900_Raw_Data_Frame_Raptor_Counts_accel_mux_tag';
    'HG9900_Raw_Data_Frame_Raptor_Counts_accel_mux_word';
    };

% gyro{N}_mux_tag
gyroTagNames = compose( ...
    strcat(options.ParamPrefix, options.ParamGyroTagTemplate), ...
    1:options.GyroCount);

gyroStructTagNames = compose( ...
    'HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_mux_tag', ...
    1:options.GyroCount);

% gyro{N}_mux_word
gyroWordNames = compose( ...
    strcat(options.ParamPrefix, options.ParamGyroWordTemplate), ...
    1:options.GyroCount);

gyroStructWordNames = compose( ...
    'HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_mux_word', ...
    1:options.GyroCount);

% gyro{N}_status
gyroStatusNames = compose( ...
    strcat(options.ParamPrefix, options.ParamGyroStatusTemplate), ...
    1:options.GyroCount);

gyroStructStatusNames = compose( ...
    'HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_status', ...
    1:options.GyroCount);

% gyro{N}_lsb
gyroThetaLsbNames = compose( ...
    strcat(options.ParamPrefix, options.ParamGyroThetaLsbTemplate), ...
    1:options.GyroCount);

gyroStructThetaLsbNames = compose( ...
    'HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_lsb', ...
    1:options.GyroCount);

% gyro{N}_msb
gyroThetaMsbNames = compose( ...
    strcat(options.ParamPrefix, options.ParamGyroThetaMsbTemplate), ...
    1:options.GyroCount);

gyroStructThetaMsbNames = compose( ...
    'HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_msb', ...
    1:options.GyroCount);

% gyro{N}_checksum
gyroChecksumNames = compose( ...
    strcat(options.ParamPrefix, options.ParamGyroChecksumTemplate), ...
    1:options.GyroCount);

gyroStructChecksumNames = compose( ...
    'HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_checksum', ...
    1:options.GyroCount);

paramNames = [
    accelNames(:);
    gyroTagNames(:);
    gyroWordNames(:);
    gyroStatusNames(:);
    gyroThetaLsbNames(:);
    gyroThetaMsbNames(:);
    gyroChecksumNames(:);
    ];

structNames = [
    accelStructNames(:);
    gyroStructTagNames(:);
    gyroStructWordNames(:);
    gyroStructStatusNames(:);
    gyroStructThetaLsbNames(:);
    gyroStructThetaMsbNames(:);
    gyroStructChecksumNames(:);
    ];

nParams = length(paramNames);
psCell = cell(nParams, 1);

for idx = 1:nParams
    paramName = paramNames{idx};

    ps = ParameterSpec(...
        'RecordingName', options.RecordingName, ...
        'VirtualChannelName', options.VirtualChannelName, ...
        'PacketName', options.PacketName, ...
        'ParameterName', paramName);

    psCell{idx} = ps;
end

parameterList = [psCell{:}]';

end
