%% Load Data from the Input Struct

absRxTime = s.AbsoluteReceivedTime;
msgNum = s.HG9900_Raw_Data_Frame_Msg_Number;
timestamp = s.HG9900_Raw_Data_Frame_Timestamp;
accelMuxTag = s.HG9900_Raw_Data_Frame_Raptor_Counts_accel_mux_tag;
accelMuxWord = s.HG9900_Raw_Data_Frame_Raptor_Counts_accel_mux_word;

%% Process the mux messages for the accels
% For each tag in the gyro mux message
for k = 1:size(accelTagDef,1)
    accelTagName    = accelTagDef{k,1};
    accelTagId      = accelTagDef{k,2};
    accelTagScale   = accelTagDef{k,3};
    accelTagUnit    = accelTagDef{k,4};
    accelTagComment = accelTagDef{k,5};
    numBytes = length(accelTagId);
    idxByte = cell(numBytes,1);
    msgNumByte = cell(numBytes,1);

    % For each byte in the mux word
    for n = 1:numBytes
        % Find all the byte locations for the current tag
        idxByte{n} = find(accelMuxTag == accelTagId(n));
        % Get the frame message number for each byte location
        msgNumByte{n} = msgNum(idxByte{n});
    end

    % Handle dropped or missing frames
    msgNumByte0 = msgNumByte{1};
    for n = 1:numBytes-1
        % Bytes should be sequential and contiguous. Strip out frames that
        % don't form a complete mux word.
        [~,ia] = intersect(msgNumByte0+n, msgNumByte{n+1}, 'stable');
        msgNumByte0 = msgNumByte0(ia);
    end

    % Determine the data type of the mux word
    switch numBytes
        case 1
            dataType = "uint8";
        case 2
            dataType = "int16";
        otherwise
            dataType = "int16";
            warning("decodeHg9900RawMux: Unexpected number of bytes for " + accelTagName)
    end

    % Stitch together bytes to form complete mux words
    idx = cell(numBytes,1);
    byte = cell(numBytes,1);
    word = 0;
    for n = 1:numBytes
        idx{n} = ismember(msgNum, msgNumByte0+n-1);
        byte{n} = bitshift(cast(accelMuxWord(idx{n}), dataType), 8*(n-1), dataType);
        word = bitor(byte{n}, word);
    end

    % Apply scaling if applicable
    if isempty(accelTagScale) || ~applyScaling
        value = double(word);
    else
        value = polyval(accelTagScale, double(word));
    end

    % Construct mux output structure
    % Assume data valid at timestamp of first byte received
    mux.accel.(accelTagName).AbsoluteReceivedTime = absRxTime(idx{1});
    mux.accel.(accelTagName).time    = timestamp(idx{1});
    %mux.accel.(accelTagName).word    = word;
    mux.accel.(accelTagName).value   = value;
    mux.accel.(accelTagName).id      = accelTagId;
    mux.accel.(accelTagName).unit    = accelTagUnit;
    mux.accel.(accelTagName).comment = accelTagComment;
end

% Generate GATV
accelTagName = 'gyro_bit_GATV';

value = bitget(value, 6);

% Generate GATW

value = bitget(value, 7);

%% Accel Mux Byte 4:5
accelTagSource = 'ie_bit';

% Generate IE_PCS
accelTagName = 'ie_bit_IE_PCS';
mux.accel.(accelTagName) = mux.accel.(accelTagSource);

% Generate IE_RRW

% Generate IE_WDT
accelTagName = 'ie_bit_IE_WDT';

value = bitget(value, 3);

% Generate APAM
accelTagName = 'ie_bit_APAM';

% Generate APAM_ST
accelTagName = 'ie_bit_APAM_ST';
mux.accel.(accelTagName) = mux.accel.(accelTagSource);
value = mux.accel.(accelTagName).value;
value = bitget(value, 6);

% Generate AETR
accelTagName = 'ie_bit_AETR';
mux.accel.(accelTagName) = mux.accel.(accelTagSource);
value = mux.accel.(accelTagName).value;
value = bitget(value, 7);

mux.accel.(accelTagName).comment = 'AE temperature sensor fail.';

% Generate ADRV
accelTagName = 'ie_bit_ADRV';
mux.accel.(accelTagName) = mux.accel.(accelTagSource);
value = mux.accel.(accelTagName).value;

% Generate AECCS

accelTagName = 'ie_bit_AEBCCS';

% Generate IE_BRRW
accelTagName = 'ie_bit_IE_BRRW';

accelTagName = 'ie_bit_IE_BPCS';

value = bitget(value, 15);
mux.accel.(accelTagName).value = value;
mux.accel.(accelTagName).comment = 'IE background program checksum fail.';

%% Process the mux messages for each gyro
for g = 1:3
    gyroMuxTag     = s.(sprintf('HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_mux_tag', g));
    gyroMuxWord    = s.(sprintf('HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_mux_word', g));
    gyroStatusByte = s.(sprintf('HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_status', g));

    gyroThetaLsb = s.(sprintf('HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_lsb', g));
    gyroThetaMsb = s.(sprintf('HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_msb', g));
    gyroChecksum = s.(sprintf('HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_checksum', g));

    % Store for current gyro so original data is not modified
    absRxTimeMux = absRxTime;
    timestampMux = timestamp;
    msgNumMux = msgNum;

    % Calculate the expected crc
    % Reference: Honeywell GG1320AN Dig-Gyro RLG User Interface Document
    % See Figure 2
    gyroRawSum = uint16(...
        gyroStatusByte ...
        + gyroMuxTag ...
        + gyroThetaLsb ...
        + gyroThetaMsb ...
        + gyroMuxWord);

    gyroSum = uint8(bitand(gyroRawSum, uint16(0xFF))); % Get the lower 8-bits
    checksumExpected = bitcmp(gyroSum);

    % Remove data that does not pass checksum
    if options.IsChecksumEnabled
        isValid = gyroChecksum == checksumExpected;
    else
        isValid = true(size(gyroMuxTag));
    end

    % Remove data that does not pass checksum
    % If disabled, then the valid flag will be true and no data will be removed
    absRxTimeMux(~isValid) = [];
    timestampMux(~isValid) = [];
    msgNumMux(~isValid) = [];
    gyroMuxTag(~isValid) = [];
    gyroMuxWord(~isValid) = [];
    gyroStatusByte(~isValid) = [];

    % Store the status
    gyroTagName = 'status_raw';
    mux.gyro(g).(gyroTagName).AbsoluteReceivedTime = absRxTimeMux;
    mux.gyro(g).(gyroTagName).time    = timestampMux;
    %mux.gyro(g).(gyroTagName).word    = uint8(gyroStatusByte);
    mux.gyro(g).(gyroTagName).value   = uint8(gyroStatusByte);
    mux.gyro(g).(gyroTagName).id      = NaN;
    mux.gyro(g).(gyroTagName).unit    = 'counts';
    mux.gyro(g).(gyroTagName).comment = 'Gyro raw status byte.';

    % For each tag in the gyro mux message
    for k = 1:size(tagDef,1)
        gyroTagName    = tagDef{k,1};
        gyroTagId      = tagDef{k,2};
        numBytes       = tagDef{k,3};
        gyroTagScale   = tagDef{k,4};
        gyroTagUnit    = tagDef{k,5};
        gyroTagComment = tagDef{k,6};
        idxByte = cell(numBytes,1);
        msgNumByte = cell(numBytes,1);

        % For each byte in the mux word
        for n = 1:numBytes
            % Find all the byte locations for the current tag
            idxByte{n} = find(gyroMuxTag == gyroTagId & ...
                bitand(gyroStatusByte, STATUS_BYTE_CNT_MASK) == (n-1));

            % Get the frame message number for each byte location
            msgNumByte{n} = msgNumMux(idxByte{n});
        end

        % Handle dropped or missing frames
        msgNumByte0 = msgNumByte{1};
        for n = 1:numBytes-1
            % Bytes should be sequential and contiguous. Strip out frames that
            % don't form a complete mux word.
            [~,ia] = intersect(msgNumByte0+n, msgNumByte{n+1}, 'stable');
            msgNumByte0 = msgNumByte0(ia);
        end

        % Determine the data type of the mux word
        switch numBytes
            case 2
                dataType = "int16";
            case 4
                dataType = "int32";
            otherwise
                dataType = "int16";
                warning("decodeHg9900RawMux: Unexpected number of bytes for " + gyroTagName)
        end

        % Stitch together bytes to form complete mux words
        idx = cell(numBytes,1);
        byte = cell(numBytes,1);
        word = 0;
        for n = 1:numBytes
            idx{n} = ismember(msgNumMux, msgNumByte0+n-1);
            byte{n} = bitshift(cast(gyroMuxWord(idx{n}), dataType), 8*(n-1), dataType);
            word = bitor(byte{n}, word);
        end

        % Apply scaling if applicable
        if isempty(gyroTagScale) || ~applyScaling
            value = double(word);
        else
            value = polyval(gyroTagScale, double(word));
        end

        % Construct mux output structure
        % Assume data valid at timestamp of first byte received
        mux.gyro(g).(gyroTagName).AbsoluteReceivedTime = absRxTimeMux(idx{1});
        mux.gyro(g).(gyroTagName).time    = timestampMux(idx{1});
        %mux.gyro(g).(gyroTagName).word    = word;
        mux.gyro(g).(gyroTagName).value   = value;
        mux.gyro(g).(gyroTagName).id      = gyroTagId;
        mux.gyro(g).(gyroTagName).unit    = gyroTagUnit;
        mux.gyro(g).(gyroTagName).comment = gyroTagComment;
    end
end

%% Cleanup Gyro Data

for g = 1:3
    %% software_rev
    % The software_rev value is contained in bits 11:0
    % bits 15:12 contain the patch
    mux.gyro(g).software_rev_core = mux.gyro(g).software_rev;
    mux.gyro(g).software_rev_patch1 = mux.gyro(g).software_rev;

    value = mux.gyro(g).software_rev_patch3.value;

    mux.gyro(g).software_rev_patch3.value = value;

    %% Generate software_rev_patch4
    value = mux.gyro(g).software_rev_patch4.value;
    value = bitget(value, 13);
    mux.gyro(g).software_rev_patch4.value = value;

    %% Generate gyro_status
    gyroTagName = 'gyro_status';
    mux.gyro(g).(gyroTagName) = mux.gyro(g).status_raw;
    value = mux.gyro(g).(gyroTagName).value;
    value = bitand(value, uint8(STATUS_BYTE_GYRO_MASK));
    value = bitshift(value, -6);
    mux.gyro(g).(gyroTagName).value = value;
    mux.gyro(g).(gyroTagName).comment = 'Gyro status byte.';

    %% Generate bit_readout_flg
    gyroTagName = 'bit_readout_flg';
    mux.gyro(g).(gyroTagName) = mux.gyro(g).status_raw;
    value = mux.gyro(g).(gyroTagName).value;
    value = bitand(value, uint8(STATUS_BYTE_READOUT_MASK));
    value = bitshift(value, -5);
    mux.gyro(g).(gyroTagName).value   = value;
    mux.gyro(g).(gyroTagName).comment = 'Readout BIT status.';

    %% Generate bit_curr_ctrl_flg
    gyroTagName = 'bit_curr_ctrl_flg';
    mux.gyro(g).(gyroTagName) = mux.gyro(g).status_raw;
    value = mux.gyro(g).(gyroTagName).value;
    value = bitand(value, uint8(STATUS_BYTE_CURCTRL_MASK));
    value = bitshift(value, -4);
    mux.gyro(g).(gyroTagName).value   = value;
    mux.gyro(g).(gyroTagName).comment = 'Current Control BIT status.';

    %% Generate bit_dither_flg
    gyroTagName = 'bit_dither_flg';
    mux.gyro(g).(gyroTagName) = mux.gyro(g).status_raw;
    value = mux.gyro(g).(gyroTagName).value;
    value = bitand(value, uint8(STATUS_BYTE_DITHER_MASK));
    value = bitshift(value, -3);
    mux.gyro(g).(gyroTagName).value   = value;
    mux.gyro(g).(gyroTagName).comment = 'Dither BIT status.';

    %% Generate bit_plc_flg
    gyroTagName = 'bit_plc_flg';
    mux.gyro(g).(gyroTagName) = mux.gyro(g).status_raw;
    value = mux.gyro(g).(gyroTagName).value;
    value = bitand(value, uint8(STATUS_BYTE_PLC_MASK));
    value = bitshift(value, -2);
    mux.gyro(g).(gyroTagName).value   = value;
    mux.gyro(g).(gyroTagName).comment = 'PLC BIT status.';

end

end
