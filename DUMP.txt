#!/usr/bin/env python2
# Make sure this is the version of Python you want to
# run this script. SimWB supports all versions of Python
# from 2.6 and newer.

# A couple of standard modules we will use. Add what you need.
import sys
import os

# This module insures that we will load the appropriate
# SimWB modules for the version of Python that is running
# this script.
import simwbPath

# This is the real-time Python module.
import simwbRT
# Besides the wealth of functionality provided by Python,
# there are other methods in the simwbRT module to give
# access (hooks) into SimWB. See "pydoc simwbRT" for details of
# all the calls.
#   logMessage            display a message to the operator
#   getFrameRate          in Hz
#   getFrameLength        in microseconds
#   getFrameCount         0,1,2... incremented every frame
#   getCVTValue           read an RTDB value by name
#   setCVTValue           write an RTDB value by name
#     (and other calls to provide the same functionality but
#      using different RTDB access methods)
#   getAltValue           read an RTDB alternate value by name
#   setAltValue           write an RTDB alternate value by name
#   getRunTimeFlags       fetch an RTDB items flags (alt, opr, etc)
#   setOperatorFlag       set an RTDB items operator flag
#   setAlternateValueFlag set an RTDB items alternate value flag
#   writeStdOut           send output to the script psuedo-stdout
#   getRTDBName           return the name of the tests RTDB
#   getInitialConditionsName
#                         return the name of a tests initial conditions set
#   getTestSession        return a dictionary with test and session name
#   testStop              tell the scheduler to stop the test
#   getItemMeta           return a dictionary of an RTDB items meta info
#                           (EU conversion, description, etc)
#   addGenerator          create a signal generator and associate with RTDB item
#                           (and associated calls to configure them)
#   startGenerator        start a single generator
#   startAllGenerators    start all generators
#   stopGenerator         stop a single generator
#   stopAllGenerators     stop all generators
#
# These routines are not exhaustive with respect to those
# available via simwbClient, but is a subset choosen by us to
# be most useful to the real-time user.

# This call connects you to the RTDB. You are free to read/write
# RTDB variables after this.
simwbRT.mapRTDB()

# These calls connect you to the SimWB message logger. You
# can send messages via simwbRT.logMessage() now.
simwbRT.initLog(os.path.basename(sys.argv[1]));

# The mapRTDB() and initLog() calls can be in either order.

# We now need to attach to the SimWB scheduler. The scheduler will
# coordinate when this task runs after this point, and it is
# crucial to the proper timing of this script.
#
# How you do this depends on if this is an initialization script
# or a regular real-time script.  An initialization script executes
# once (prior to any models, but after everything is initialized)
# and then goes away.  A regular real-time script runs every cycle
# (after input tasks but before any models) and keeps running until
# the scheduler ends the test.
# Attach to the scheduler as a real-time script.
simwbRT.initSched(sys.argv[1])

# To access some useful/needed constants.
import simwbConstants



try:
    # Scheduler init cycles.
    for i in range(simwbConstants.SCHED_INITCYCLES):
        # We block on this call until freed by the scheduler.
        simwbRT.waitNextFrame()

    # Enter the real-time loop.
    while True:
        simwbRT.waitNextFrame()
        # Do your work here


except KeyboardInterrupt:
    # The scheduler sends tasks a SIGINT (in Python that's a
    # KeyboardInterrupt) to signal the test is complete. We trap
    # it for a clean shutdown.
    #
    # You can add any shutdown code here.
    #
    sys.exit(0)
