%$Insert-OUO-Block
function [parameterList, structNames] = getHg9900ParameterList(options)
% function [parameterList, structNames] = getHg9900ParameterList(options)
%
% Gets the list of parameters to properly read the raw IMU data.
%
% NAME/VALUE ARGUMENTS
% --------------------
% RecordingName
% VirtualChannelName
% PacketName
% ParamPrefix
% GyroCount
% ParamMsgNumber
% ParamTimestamp
% ParamAccelTag
% ParamAccelWord
% ParamGyroTagTemplate
% ParamGyroWordTemplate
% ParamGyroStatusStatusTemplate
% ParamGyroReadoutBitStatusTemplate
% ParamGyroCurrentBitStatusTemplate
% ParamGyroDitherBitStatusTemplate
% ParamGyroPlcBitStatusTemplate
% ParamGyroStatusIdCounterTemplate
% ParamGyroThetaLsbTemplate
% ParamGyroThetaMsbTemplate
% ParamGyroChecksumTemplate


arguments
    % Name of the TM recording.
    options.RecordingName char {mustBeTextScalar} = '';

    % Name of the CCSDS Virtual Channel.
    options.VirtualChannelName char ...
        {mustBeTextScalar} = 'Navigation';

    % Name of the CCSDS Space Packet.
    options.PacketName char ...
        {mustBeTextScalar} = ...
        ['...MinAtr...IMU...HG9900...Space_Packet' ...
         '--HG9900_Raw_Data'];

    % A prefix added to the ParameterName when loading from the packet.
    options.ParamPrefix char ...
        {mustBeTextScalar} = '';

    % The number of gyrometer channels available within the IMU.
    options.GyroCount(1,1) double = 3;

    % The parameter name of the message number.
    options.ParamMsgNumber char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
         'HG9900_Raw_Data_Frame--Msg_Number'];

    % The parameter name of the time stamp.
    options.ParamTimestamp char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
         'HG9900_Raw_Data_Frame--Time_Of_Validity'];

    % The parameter name of the accelerometer tag.
    options.ParamAccelTag char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
         'HG9900_Raw_Data_Frame--Raptor_Counts--accel_mux_tag'];

    % The parameter name of the accelerometer word.
    options.ParamAccelWord char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
         'HG9900_Raw_Data_Frame--Raptor_Counts--accel_mux_word'];


    %% ============================================================
    %  Gyro mux fields
    %  ============================================================

    % The sprintf template for the parameter name of the gyrometer tag.
    options.ParamGyroTagTemplate char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
         'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%s--mux_tag'];

    % The sprintf template for the parameter name of the gyrometer word.
    options.ParamGyroWordTemplate char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
         'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%s--mux_word'];


    %% ============================================================
    %  Split gyro status fields
    %  ============================================================

    % Gyrometer STATUS field.
    options.ParamGyroStatusStatusTemplate char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
         'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%s--status--status'];

    % Gyrometer readout BIT status.
    options.ParamGyroReadoutBitStatusTemplate char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
         'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%s--status--readout_bit_status'];

    % Gyrometer current BIT status.
    options.ParamGyroCurrentBitStatusTemplate char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
         'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%s--status--current_bit_status'];

    % Gyrometer dither BIT status.
    options.ParamGyroDitherBitStatusTemplate char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
         'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%s--status--dither_bit_status'];

    % Gyrometer PLC BIT status.
    options.ParamGyroPlcBitStatusTemplate char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
         'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%s--status--plc_bit_status'];

    % Gyrometer status ID counter.
    options.ParamGyroStatusIdCounterTemplate char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
         'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%s--status--status_id_counter'];


    %% ============================================================
    %  Theta / checksum fields
    %  ============================================================

    % The sprintf template for the parameter name of the gyrometer
    % theta LSB.
    options.ParamGyroThetaLsbTemplate char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
         'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%s--lsb'];

    % The sprintf template for the parameter name of the gyrometer
    % theta MSB.
    options.ParamGyroThetaMsbTemplate char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
         'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%s--msb'];

    % The sprintf template for the parameter name of the gyrometer
    % checksum.
    options.ParamGyroChecksumTemplate char ...
        {mustBeTextScalar} = ...
        ['/MinAtr/IMU/HG9900/' ...
         'HG9900_Raw_Data_Frame--Raptor_Counts--gyro%s--checksum'];
end


import MinAtrTmAnalysis.Framework.Data.ParameterSpec


%% ================================================================
%  Common / accelerometer parameters
%  ================================================================

accelNames = {
    options.ParamMsgNumber;
    options.ParamTimestamp;
    options.ParamAccelTag;
    options.ParamAccelWord;
};

accelStructNames = {
    'HG9900_Raw_Data_Frame_Msg_Number';
    'HG9900_Raw_Data_Frame_Timestamp';
    'HG9900_Raw_Data_Frame_Raptor_Counts_accel_mux_tag';
    'HG9900_Raw_Data_Frame_Raptor_Counts_accel_mux_word';
};


%% ================================================================
%  Gyro parameter names
%  ================================================================

% External telemetry axes:
%
%   U -> gyro1
%   V -> gyro2
%   W -> gyro3
%
gyroAxes = ["U", "V", "W"];
gyroIndices = 1:options.GyroCount;


%% ================================================================
%  Gyro mux tag
%  ================================================================

gyroTagNames = cellstr(compose( ...
    strcat(options.ParamPrefix, options.ParamGyroTagTemplate), ...
    gyroAxes));

gyroStructTagNames = cellstr(compose( ...
    'HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_mux_tag', ...
    gyroIndices));


%% ================================================================
%  Gyro mux word / byte
%  ================================================================
%
% Preserve mux_word internally for decoder compatibility.

gyroWordNames = cellstr(compose( ...
    strcat(options.ParamPrefix, options.ParamGyroWordTemplate), ...
    gyroAxes));

gyroStructWordNames = cellstr(compose( ...
    'HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_mux_word', ...
    gyroIndices));


%% ================================================================
%  Gyro STATUS
%  ================================================================
%
% This is now the new split 2-bit Status field.
%
% Do NOT map this to:
%
%   gyro%d_status
%
% because that historically represented the complete packed
% 8-bit status byte.

gyroStatusNames = cellstr(compose( ...
    strcat(options.ParamPrefix, ...
           options.ParamGyroStatusStatusTemplate), ...
    gyroAxes));

gyroStructStatusNames = cellstr(compose( ...
    ['HG9900_Raw_Data_Frame_Raptor_Counts_' ...
     'gyro%d_status_status'], ...
    gyroIndices));


%% ================================================================
%  Gyro readout BIT status
%  ================================================================

gyroReadoutBitStatusNames = cellstr(compose( ...
    strcat(options.ParamPrefix, ...
           options.ParamGyroReadoutBitStatusTemplate), ...
    gyroAxes));

gyroStructReadoutBitStatusNames = cellstr(compose( ...
    ['HG9900_Raw_Data_Frame_Raptor_Counts_' ...
     'gyro%d_status_readout_bit_status'], ...
    gyroIndices));


%% ================================================================
%  Gyro current BIT status
%  ================================================================

gyroCurrentBitStatusNames = cellstr(compose( ...
    strcat(options.ParamPrefix, ...
           options.ParamGyroCurrentBitStatusTemplate), ...
    gyroAxes));

gyroStructCurrentBitStatusNames = cellstr(compose( ...
    ['HG9900_Raw_Data_Frame_Raptor_Counts_' ...
     'gyro%d_status_current_bit_status'], ...
    gyroIndices));


%% ================================================================
%  Gyro dither BIT status
%  ================================================================

gyroDitherBitStatusNames = cellstr(compose( ...
    strcat(options.ParamPrefix, ...
           options.ParamGyroDitherBitStatusTemplate), ...
    gyroAxes));

gyroStructDitherBitStatusNames = cellstr(compose( ...
    ['HG9900_Raw_Data_Frame_Raptor_Counts_' ...
     'gyro%d_status_dither_bit_status'], ...
    gyroIndices));


%% ================================================================
%  Gyro PLC BIT status
%  ================================================================

gyroPlcBitStatusNames = cellstr(compose( ...
    strcat(options.ParamPrefix, ...
           options.ParamGyroPlcBitStatusTemplate), ...
    gyroAxes));

gyroStructPlcBitStatusNames = cellstr(compose( ...
    ['HG9900_Raw_Data_Frame_Raptor_Counts_' ...
     'gyro%d_status_plc_bit_status'], ...
    gyroIndices));


%% ================================================================
%  Gyro status ID counter
%  ================================================================

gyroStatusIdCounterNames = cellstr(compose( ...
    strcat(options.ParamPrefix, ...
           options.ParamGyroStatusIdCounterTemplate), ...
    gyroAxes));

gyroStructStatusIdCounterNames = cellstr(compose( ...
    ['HG9900_Raw_Data_Frame_Raptor_Counts_' ...
     'gyro%d_status_id_counter'], ...
    gyroIndices));


%% ================================================================
%  Gyro theta LSB
%  ================================================================
%
% Preserve the old internal lsb name for decoder compatibility.

gyroThetaLsbNames = cellstr(compose( ...
    strcat(options.ParamPrefix, ...
           options.ParamGyroThetaLsbTemplate), ...
    gyroAxes));

gyroStructThetaLsbNames = cellstr(compose( ...
    'HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_lsb', ...
    gyroIndices));


%% ================================================================
%  Gyro theta MSB
%  ================================================================

gyroThetaMsbNames = cellstr(compose( ...
    strcat(options.ParamPrefix, ...
           options.ParamGyroThetaMsbTemplate), ...
    gyroAxes));

gyroStructThetaMsbNames = cellstr(compose( ...
    'HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_msb', ...
    gyroIndices));


%% ================================================================
%  Gyro checksum
%  ================================================================
%
% Preserve the existing internal checksum name for now.
% Checksum behavior itself will be handled later.

gyroChecksumNames = cellstr(compose( ...
    strcat(options.ParamPrefix, ...
           options.ParamGyroChecksumTemplate), ...
    gyroAxes));

gyroStructChecksumNames = cellstr(compose( ...
    'HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_checksum', ...
    gyroIndices));


%% ================================================================
%  Complete raw parameter list
%  ================================================================

paramNames = [
    accelNames(:);

    gyroTagNames(:);
    gyroWordNames(:);

    gyroStatusNames(:);
    gyroReadoutBitStatusNames(:);
    gyroCurrentBitStatusNames(:);
    gyroDitherBitStatusNames(:);
    gyroPlcBitStatusNames(:);
    gyroStatusIdCounterNames(:);

    gyroThetaLsbNames(:);
    gyroThetaMsbNames(:);
    gyroChecksumNames(:);
];


%% ================================================================
%  Corresponding internal struct field names
%  ================================================================

structNames = [
    accelStructNames(:);

    gyroStructTagNames(:);
    gyroStructWordNames(:);

    gyroStructStatusNames(:);
    gyroStructReadoutBitStatusNames(:);
    gyroStructCurrentBitStatusNames(:);
    gyroStructDitherBitStatusNames(:);
    gyroStructPlcBitStatusNames(:);
    gyroStructStatusIdCounterNames(:);

    gyroStructThetaLsbNames(:);
    gyroStructThetaMsbNames(:);
    gyroStructChecksumNames(:);
];


%% ================================================================
%  Build ParameterSpec list
%  ================================================================

nParams = length(paramNames);
psCell = cell(nParams, 1);

for idx = 1:nParams

    paramName = paramNames{idx};

    ps = ParameterSpec( ...
        'RecordingName', options.RecordingName, ...
        'VirtualChannelName', options.VirtualChannelName, ...
        'PacketName', options.PacketName, ...
        'ParameterName', paramName);

    psCell{idx} = ps;
end

parameterList = [psCell{:}]';

end
