import numpy as np
from scipy.io import savemat

trajectory_rate = 20.0
trajectory_duration = 60.0

dt = 1.0 / trajectory_rate
trajectory_time = np.arange(
    0.0,
    trajectory_duration + dt,
    dt,
)

north_velocity = 10.0

position_north = north_velocity * trajectory_time
position_east = 50.0 * np.sin(0.1 * trajectory_time)
position_down = np.zeros_like(trajectory_time)

velocity_north = np.full_like(
    trajectory_time,
    north_velocity,
)
velocity_east = 5.0 * np.cos(0.1 * trajectory_time)
velocity_down = np.zeros_like(trajectory_time)

savemat(
    "gps_trajectory.mat",
    {
        "trajectory_time": trajectory_time,
        "position_north": position_north,
        "position_east": position_east,
        "position_down": position_down,
        "velocity_north": velocity_north,
        "velocity_east": velocity_east,
        "velocity_down": velocity_down,
    },
)
