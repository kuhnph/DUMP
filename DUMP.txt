#!/usr/bin/env python2

import sys
import os
import csv
import bisect

import simwbPath
import simwbRT
import simwbConstants


# =============================================================================
# RTDB VARIABLE NAMES
# =============================================================================

TIME_RTDB   = "Spirent.SW_T_Simulation"

X_ECEF_RTDB = "Spirent.SW_Xecef"
V_ECEF_RTDB = "Spirent.SW_Vecef"
A_ECEF_RTDB = "Spirent.SW_Aecef"
J_ECEF_RTDB = "Spirent.SW_Jecef"
EULER_RTDB  = "Spirent.SW_Euler"


# =============================================================================
# TRAJECTORY FILE
# =============================================================================

# Put firefly_trac.csv in the same directory as this Python model.
MODEL_DIRECTORY = os.path.dirname(os.path.abspath(__file__))

CSV_FILE = os.path.join(
    MODEL_DIRECTORY,
    "firefly_trac.csv"
)


# =============================================================================
# EXPECTED CSV FORMAT
# =============================================================================
#
# Column 0:
#     time
#
# Columns 1-3:
#     X_ECEF_1
#     X_ECEF_2
#     X_ECEF_3
#
# Columns 4-6:
#     V_ECEF_1
#     V_ECEF_2
#     V_ECEF_3
#
# Columns 7-9:
#     A_ECEF_1
#     A_ECEF_2
#     A_ECEF_3
#
# Columns 10-12:
#     J_ECEF_1
#     J_ECEF_2
#     J_ECEF_3
#
# Columns 13-15:
#     yaw
#     pitch
#     roll
#
# Total:
#     16 columns
#
# =============================================================================


class Trajectory(object):

    def __init__(self, filename):

        self.time = []
        self.data = []

        self.load_csv(filename)

        if len(self.time) < 2:

            raise RuntimeError(
                "Trajectory file must contain at least two valid rows."
            )


        # Verify monotonically increasing time
        for i in range(1, len(self.time)):

            if self.time[i] <= self.time[i - 1]:

                raise RuntimeError(
                    "Trajectory time must be strictly increasing."
                )


    def load_csv(self, filename):

        f = open(filename, "r")

        try:

            reader = csv.reader(f)

            for row in reader:

                # Ignore malformed rows
                if len(row) != 16:
                    continue


                try:

                    values = [
                        float(value.strip())
                        for value in row
                    ]

                except ValueError:

                    # Allows header rows to be ignored
                    continue


                # First column is time
                self.time.append(
                    values[0]
                )


                # Remaining 15 columns are trajectory values
                self.data.append(
                    values[1:]
                )

        finally:

            f.close()


    def sample(self, t):

        # ---------------------------------------------------------------------
        # Before trajectory begins:
        # hold first point
        # ---------------------------------------------------------------------

        if t <= self.time[0]:

            return self.data[0]


        # ---------------------------------------------------------------------
        # After trajectory ends:
        # hold final point
        # ---------------------------------------------------------------------

        if t >= self.time[-1]:

            return self.data[-1]


        # ---------------------------------------------------------------------
        # Find:
        #
        # time[i] <= t < time[i + 1]
        # ---------------------------------------------------------------------

        i = bisect.bisect_right(
            self.time,
            t
        ) - 1


        t0 = self.time[i]
        t1 = self.time[i + 1]

        y0 = self.data[i]
        y1 = self.data[i + 1]


        # ---------------------------------------------------------------------
        # Linear interpolation fraction
        # ---------------------------------------------------------------------

        alpha = (
            (t - t0)
            /
            (t1 - t0)
        )


        # ---------------------------------------------------------------------
        # Interpolate all 15 trajectory outputs
        # ---------------------------------------------------------------------

        values = []

        for j in range(15):

            value = (
                y0[j]
                +
                alpha * (y1[j] - y0[j])
            )

            values.append(
                value
            )


        return values


# =============================================================================
# CONNECT TO SIMWB
# =============================================================================

simwbRT.mapRTDB()

simwbRT.initLog(
    os.path.basename(sys.argv[1])
)


# =============================================================================
# VERIFY RTDB
# =============================================================================

simwbRT.logMessage(
    "trajectoryLookUp RTDB name = %s"
    % str(simwbRT.getRTDBName())
)


# Verify the time input exists
initial_time = simwbRT.getCVTValue(
    TIME_RTDB
)

simwbRT.logMessage(
    "%s initial value = %s"
    % (
        TIME_RTDB,
        str(initial_time)
    )
)


# Verify the output variables exist
initial_x = simwbRT.getCVTValue(
    X_ECEF_RTDB
)

initial_v = simwbRT.getCVTValue(
    V_ECEF_RTDB
)

initial_a = simwbRT.getCVTValue(
    A_ECEF_RTDB
)

initial_j = simwbRT.getCVTValue(
    J_ECEF_RTDB
)

initial_euler = simwbRT.getCVTValue(
    EULER_RTDB
)


simwbRT.logMessage(
    "X_ECEF initial = %s"
    % str(initial_x)
)

simwbRT.logMessage(
    "V_ECEF initial = %s"
    % str(initial_v)
)

simwbRT.logMessage(
    "A_ECEF initial = %s"
    % str(initial_a)
)

simwbRT.logMessage(
    "J_ECEF initial = %s"
    % str(initial_j)
)

simwbRT.logMessage(
    "Euler initial = %s"
    % str(initial_euler)
)


# =============================================================================
# LOAD TRAJECTORY
# =============================================================================

try:

    trajectory = Trajectory(
        CSV_FILE
    )


    simwbRT.logMessage(
        "Loaded trajectory: %d samples, %.6f to %.6f seconds"
        % (
            len(trajectory.time),
            trajectory.time[0],
            trajectory.time[-1]
        )
    )


except Exception as error:

    simwbRT.logMessage(
        "Trajectory load failed: %s"
        % str(error)
    )

    simwbRT.initFailed()

    sys.exit(1)


# =============================================================================
# ATTACH TO SIMWB SCHEDULER
# =============================================================================

simwbRT.initSched(
    sys.argv[1]
)


# =============================================================================
# SCHEDULER INITIALIZATION CYCLES
# =============================================================================

try:

    for i in range(
        simwbConstants.SCHED_INITCYCLES
    ):

        simwbRT.waitNextFrame()


    simwbRT.logMessage(
        "trajectoryLookUp entering real-time loop"
    )


    # =========================================================================
    # REAL-TIME LOOP
    # =========================================================================

    while True:

        # ---------------------------------------------------------------------
        # Wait for SimWB scheduler
        # ---------------------------------------------------------------------

        simwbRT.waitNextFrame()


        # ---------------------------------------------------------------------
        # Read simulation time produced by pyTime
        # ---------------------------------------------------------------------

        time_value = simwbRT.getCVTValue(
            TIME_RTDB
        )


        # If time cannot be read, skip this frame
        if time_value is None:

            continue


        t = float(
            time_value[0]
        )


        # ---------------------------------------------------------------------
        # Interpolate trajectory at current simulation time
        # ---------------------------------------------------------------------

        values = trajectory.sample(
            t
        )


        # ---------------------------------------------------------------------
        # Map trajectory values into RTDB vectors
        # ---------------------------------------------------------------------

        X_ECEF = [
            values[0],
            values[1],
            values[2]
        ]


        V_ECEF = [
            values[3],
            values[4],
            values[5]
        ]


        A_ECEF = [
            values[6],
            values[7],
            values[8]
        ]


        J_ECEF = [
            values[9],
            values[10],
            values[11]
        ]


        Euler = [
            values[12],
            values[13],
            values[14]
        ]


        # ---------------------------------------------------------------------
        # Write trajectory to shared SimWB RTDB
        # ---------------------------------------------------------------------

        simwbRT.setCVTValue(
            X_ECEF_RTDB,
            X_ECEF
        )


        simwbRT.setCVTValue(
            V_ECEF_RTDB,
            V_ECEF
        )


        simwbRT.setCVTValue(
            A_ECEF_RTDB,
            A_ECEF
        )


        simwbRT.setCVTValue(
            J_ECEF_RTDB,
            J_ECEF
        )


        simwbRT.setCVTValue(
            EULER_RTDB,
            Euler
        )


except KeyboardInterrupt:

    simwbRT.logMessage(
        "trajectoryLookUp shutting down"
    )

    sys.exit(0)
