#define HOSTIS64BIT        1
#define HAS32BITAPILICENSE 2
typedef struct RTDBMetaInfoShared {
    char simwb_build[64];  /**< SimWB string build version identifier. */
    int  hashTableSize;      /**< The size of the hash table necessary to access the RTDB points.
                              It should be at least twice the maximum number of RTDB items used in order
                              to limit collisions in the has table and obtain good performance */
    int  maxNumRTDBItems;   /* Determines the size of named item hashtable */
    int  maxCVTSize;        /* The size of the CVT and alternate tables. */
    int  nextItemIndex;

    int  numRTDBItemsDefined;    /**< Total number of RTDB items defined in the real-time database */
    int  numCVTEntries;
    int  numHardwareConfigsDefined;
    int  spareA[3];
    int  maxNumStringVariables;  /**< Maximum # of string variables we can store in the RTDB */
    int  RTDBLoaded;             /**< Wheter a Data base has been loaded in memory. */
    int  frameLength;            /**< Frame length in micro seconds the FBS has been programmed to run at */
    unsigned int frameCount;    /**< Current frame count when CCURSIM is running. */ 
    int  running;                /**< Whether the CCURSIM simulator is running. */
    int  dLoggerRunning;         /**< Wheter the data logger is running.*/
    int  circularDLoggerRunning; /**< Wheter the circular data logger is running.*/
    char dbFileName[MAX_RTDBLOAD][MAX_DBNAME];  /**< db name of loaded RTDB */
    int  cvtDBOffset[MAX_RTDBLOAD];    /**< Offset for the RTDB in memory. */
    DBBoard dbBoard[MAX_SIMWBBOARDS];  /**<  db association between board and RTDB name that is loaded */ 
    int  nRTDBs;                  /** < Number of RTDB loaded */
    char dbDescription[MAX_DBDESCRIPTION]; /**< description of the currently load database. */
    char testId[MAX_TESTID];    /**< Test id for currently running test if any.*/
    char sessionId[MAX_SESSIONID];  /**< Session id for currently running test if any.*/
    char initCondName[MAX_INITCONDNAME]; /**< Name of the initial conditions set used during this run.*/
    unsigned int dbCheckSum;    /**< RTDB check sum to detect when a new DB has been loaded.*/
    unsigned int nextAvailCVTOffset;
    int  cvtOffsetLock;
    int  circLoggerLockBehavior ; /**< How the circular data logger is to handle the lock. One of CIRLOGGER_XXX*/
    int  dataLoggerDirectIO;    
    int  dataLoggerEnabled;    
    int  schedStats;            /**< Whether to collect scheduling statistics. */
    char loggedUserName[32];    /**< User that starts the test. */
    int  boardLock  ;
    int  rtdbLock  ;
    int  rtdbBigLock  ;
    pid_t cfgsrvPid;
    char selectedProject[MAX_PROJECTNAME]  ;
    char snapshotNameTemplate[PATH_MAX];
    char snapshotNameMask[MAX_TAGLEN];
    int  snapshotFormat;
    int  snapshotMask;
    int  takeSnapshot;
    int  flags;
    unsigned long long tsTestStarted __attribute__ ((aligned(8))) ;
    char sessionUserParm[MAX_SESSIONUSERPARM]; /**<  user parameter specified fot the running session */
} RTDBMetaInfoShared;
