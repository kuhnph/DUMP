#!/usr/bin/env python2

import sys
import os
import csv
import bisect

import simwbPath
import simwbRT
import simwbConstants


# =============================================================================
# RTDB VARIABLE NAMES
# =============================================================================

TIME_RTDB   = "Spirent.SW_T_Simulation"

X_ECEF_RTDB = "Spirent.SW_Xecef"
V_ECEF_RTDB = "Spirent.SW_Vecef"
A_ECEF_RTDB = "Spirent.SW_Aecef"
J_ECEF_RTDB = "Spirent.SW_Jecef"
EULER_RTDB  = "Spirent.SW_Euler"


# =============================================================================
# TRAJECTORY FILE
# =============================================================================

MODEL_DIRECTORY = os.path.dirname(
    os.path.abspath(__file__)
)

CSV_FILE = os.path.join(
    MODEL_DIRECTORY,
    "firefly_trac.csv"
)


# =============================================================================
# TRAJECTORY CLASS
# =============================================================================

class Trajectory(object):

    def __init__(self, filename):

        self.time = []
        self.data = []

        self.load_csv(filename)


        if len(self.time) < 2:

            raise RuntimeError(
                "Trajectory file must contain at least two valid rows."
            )


        # Verify strictly increasing trajectory time
        for i in range(1, len(self.time)):

            if self.time[i] <= self.time[i - 1]:

                raise RuntimeError(
                    "Trajectory time must be strictly increasing."
                )


    def load_csv(self, filename):

        f = open(filename, "r")

        try:

            reader = csv.reader(f)

            for row in reader:

                # Expected:
                #
                # time
                # X_ECEF[3]
                # V_ECEF[3]
                # A_ECEF[3]
                # J_ECEF[3]
                # Euler[3]
                #
                # = 16 total columns

                if len(row) != 16:
                    continue


                try:

                    values = [
                        float(value.strip())
                        for value in row
                    ]

                except ValueError:

                    # Ignore header / nonnumeric rows
                    continue


                self.time.append(
                    values[0]
                )

                self.data.append(
                    values[1:]
                )

        finally:

            f.close()


    def sample(self, t):

        # ---------------------------------------------------------------------
        # Before beginning:
        # hold first point
        # ---------------------------------------------------------------------

        if t <= self.time[0]:

            return self.data[0]


        # ---------------------------------------------------------------------
        # After end:
        # hold final point
        # ---------------------------------------------------------------------

        if t >= self.time[-1]:

            return self.data[-1]


        # ---------------------------------------------------------------------
        # Find surrounding samples:
        #
        # time[i] <= t < time[i + 1]
        # ---------------------------------------------------------------------

        i = bisect.bisect_right(
            self.time,
            t
        ) - 1


        t0 = self.time[i]
        t1 = self.time[i + 1]

        y0 = self.data[i]
        y1 = self.data[i + 1]


        # Linear interpolation fraction
        alpha = (
            (t - t0)
            /
            (t1 - t0)
        )


        values = []

        for j in range(15):

            value = (
                y0[j]
                +
                alpha * (y1[j] - y0[j])
            )

            values.append(
                value
            )


        return values


# =============================================================================
# CONNECT TO SIMWB
# =============================================================================

simwbRT.mapRTDB()

simwbRT.initLog(
    os.path.basename(sys.argv[1])
)


# =============================================================================
# IMPORTANT:
# ATTACH TO THE SCHEDULER IMMEDIATELY
# =============================================================================

simwbRT.initSched(
    sys.argv[1]
)


# =============================================================================
# RUN
# =============================================================================

try:

    # -------------------------------------------------------------------------
    # INITIALIZATION
    #
    # We are now already attached to the SimWB scheduler.
    # -------------------------------------------------------------------------

    simwbRT.logMessage(
        "trajectoryLookUp RTDB name = %s"
        % str(simwbRT.getRTDBName())
    )


    # -------------------------------------------------------------------------
    # Load the CSV trajectory
    # -------------------------------------------------------------------------

    simwbRT.logMessage(
        "trajectoryLookUp loading trajectory file: %s"
        % CSV_FILE
    )


    try:

        trajectory = Trajectory(
            CSV_FILE
        )


    except Exception as error:

        simwbRT.logMessage(
            "Trajectory load failed: %s"
            % str(error)
        )

        simwbRT.initFailed()

        sys.exit(1)


    simwbRT.logMessage(
        "Trajectory loaded: %d samples, %.6f to %.6f seconds"
        % (
            len(trajectory.time),
            trajectory.time[0],
            trajectory.time[-1]
        )
    )


    # -------------------------------------------------------------------------
    # Verify RTDB items
    # -------------------------------------------------------------------------

    initial_time = simwbRT.getCVTValue(
        TIME_RTDB
    )

    initial_x = simwbRT.getCVTValue(
        X_ECEF_RTDB
    )

    initial_v = simwbRT.getCVTValue(
        V_ECEF_RTDB
    )

    initial_a = simwbRT.getCVTValue(
        A_ECEF_RTDB
    )

    initial_j = simwbRT.getCVTValue(
        J_ECEF_RTDB
    )

    initial_euler = simwbRT.getCVTValue(
        EULER_RTDB
    )


    simwbRT.logMessage(
        "Time initial = %s"
        % str(initial_time)
    )

    simwbRT.logMessage(
        "X_ECEF initial = %s"
        % str(initial_x)
    )

    simwbRT.logMessage(
        "V_ECEF initial = %s"
        % str(initial_v)
    )

    simwbRT.logMessage(
        "A_ECEF initial = %s"
        % str(initial_a)
    )

    simwbRT.logMessage(
        "J_ECEF initial = %s"
        % str(initial_j)
    )

    simwbRT.logMessage(
        "Euler initial = %s"
        % str(initial_euler)
    )


    # -------------------------------------------------------------------------
    # Scheduler initialization cycles
    # -------------------------------------------------------------------------

    for i in range(
        simwbConstants.SCHED_INITCYCLES
    ):

        simwbRT.waitNextFrame()


    simwbRT.logMessage(
        "trajectoryLookUp entering real-time loop"
    )


    # =========================================================================
    # REAL-TIME LOOP
    # =========================================================================

    while True:

        # Wait for next SimWB frame
        simwbRT.waitNextFrame()


        # ---------------------------------------------------------------------
        # Read simulation time generated by pyTime
        # ---------------------------------------------------------------------

        time_value = simwbRT.getCVTValue(
            TIME_RTDB
        )


        if time_value is None:

            continue


        t = float(
            time_value[0]
        )


        # ---------------------------------------------------------------------
        # Interpolate trajectory
        # ---------------------------------------------------------------------

        values = trajectory.sample(
            t
        )


        # ---------------------------------------------------------------------
        # Build RTDB vectors
        # ---------------------------------------------------------------------

        X_ECEF = [
            values[0],
            values[1],
            values[2]
        ]


        V_ECEF = [
            values[3],
            values[4],
            values[5]
        ]


        A_ECEF = [
            values[6],
            values[7],
            values[8]
        ]


        J_ECEF = [
            values[9],
            values[10],
            values[11]
        ]


        Euler = [
            values[12],
            values[13],
            values[14]
        ]


        # ---------------------------------------------------------------------
        # Write trajectory to shared RTDB
        # ---------------------------------------------------------------------

        simwbRT.setCVTValue(
            X_ECEF_RTDB,
            X_ECEF
        )

        simwbRT.setCVTValue(
            V_ECEF_RTDB,
            V_ECEF
        )

        simwbRT.setCVTValue(
            A_ECEF_RTDB,
            A_ECEF
        )

        simwbRT.setCVTValue(
            J_ECEF_RTDB,
            J_ECEF
        )

        simwbRT.setCVTValue(
            EULER_RTDB,
            Euler
        )


except KeyboardInterrupt:

    simwbRT.logMessage(
        "trajectoryLookUp shutting down"
    )

    sys.exit(0)
