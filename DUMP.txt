case 3

    % Body angular velocity
    w_bf = [0.1;
            0.2;
            0.3];

    varBus.w_bf = w_bf;

    omega = norm(w_bf);

    if omega > 0

        theta = omega*t;

        u = w_bf / omega;

        varBus.q_bf = [cos(theta/2);
                       u*sin(theta/2)];

    else

        % no rotation
        varBus.q_bf = [1;0;0;0];

    end
