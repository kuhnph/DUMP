% setup_StateBus.m
names = {'A_ECEF','V_ECEF','X_ECEF', ...
         'q_be','q_bf','q_bi', ...
         'Euler321', ...
         'w_be','w_bf','w_bi', ...
         'wDot_bi'};

elems = Simulink.BusElement.empty;

for i = 1:numel(names)
    elems(i) = Simulink.BusElement;
    elems(i).Name = names{i};
    elems(i).Dimensions = [3 1];
    elems(i).DataType = 'double';
end

StateBus = Simulink.Bus;
StateBus.Elements = elems;

assignin('base','StateBus',StateBus);
