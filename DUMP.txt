#!/usr/bin/env python2

import sys
import os

import simwbPath
import simwbRT
import simwbConstants


TIME_RTDB = "Spirent.SW_T_Simulation"


simwbRT.mapRTDB()

simwbRT.initLog(
    os.path.basename(sys.argv[1])
)


simwbRT.logMessage(
    "RTDB name = %s"
    % str(simwbRT.getRTDBName())
)


value = simwbRT.getCVTValue(
    TIME_RTDB
)

simwbRT.logMessage(
    "%s initial value = %s"
    % (
        TIME_RTDB,
        str(value)
    )
)


simwbRT.initSched(
    sys.argv[1]
)


frame_rate = float(
    simwbRT.getFrameRate()
)

dt = 1.0 / frame_rate


try:

    for i in range(simwbConstants.SCHED_INITCYCLES):
        simwbRT.waitNextFrame()


    start_frame = simwbRT.getFrameCount()


    while True:

        simwbRT.waitNextFrame()

        frame_count = simwbRT.getFrameCount()

        relative_frame = frame_count - start_frame

        t_simulation = float(relative_frame) * dt


        simwbRT.setCVTValue(
            TIME_RTDB,
            t_simulation
        )


        if relative_frame % 1200 == 0:

            readback = simwbRT.getCVTValue(
                TIME_RTDB
            )

            simwbRT.logMessage(
                "frame=%d calculated=%f RTDB readback=%s"
                % (
                    frame_count,
                    t_simulation,
                    str(readback)
                )
            )


except KeyboardInterrupt:

    simwbRT.logMessage(
        "pyTime shutting down"
    )

    sys.exit(0)
