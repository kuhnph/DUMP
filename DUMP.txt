for i = 1:nOut
    signalName = outputNames{i};

    try
        lineHandle = add_line(wrapperModelName, ...
            sprintf('%s/%d', subsystemName, i), ...
            sprintf('%s/%d', busCreatorName, i), ...
            'autorouting', 'on');

        set_param(lineHandle, 'Name', signalName);

    catch ME
        warning('Could not connect named signal "%s" to Bus Creator input %d: %s', ...
            signalName, i, ME.message);
    end
end
