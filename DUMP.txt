function testHg9900ParameterList()
% testHg9900ParameterList
%
% Verifies the raw HG9900 parameter mapping produced by
% getHg9900ParameterList.
%
% Checks:
%   - Common message/timestamp/accelerometer fields
%   - Gyro U -> gyro1
%   - Gyro V -> gyro2
%   - Gyro W -> gyro3
%   - mux tag
%   - mux word
%   - split status fields
%   - theta LSB/MSB
%   - checksum


%% ================================================================
%  GET ACTUAL MAPPING
%  ================================================================

[parameterList, structNames] = getHg9900ParameterList();


%% Extract ParameterName from each ParameterSpec object

actualParamNames = strings(numel(parameterList), 1);

for idx = 1:numel(parameterList)
    actualParamNames(idx) = string(parameterList(idx).ParameterName);
end

actualStructNames = string(structNames(:));


%% ================================================================
%  EXPECTED EXTERNAL PARAMETER NAMES
%  ================================================================

base = "/MinAtr/IMU/HG9900/";

rawBase = ...
    base + "HG9900_Raw_Data_Frame--Raptor_Counts--";

gyroAxes = ["U"; "V"; "W"];


expectedParamNames = [

    % ------------------------------------------------------------
    % Common fields
    % ------------------------------------------------------------

    base + "HG9900_Raw_Data_Frame--Msg_Number";

    base + "HG9900_Raw_Data_Frame--Time_Of_Validity";

    rawBase + "accel_mux_tag";

    rawBase + "accel_mux_word";


    % ------------------------------------------------------------
    % Gyro mux tag
    % ------------------------------------------------------------

    rawBase + "gyro" + gyroAxes + "--mux_tag";


    % ------------------------------------------------------------
    % Gyro mux word
    % ------------------------------------------------------------

    rawBase + "gyro" + gyroAxes + "--mux_word";


    % ------------------------------------------------------------
    % Gyro STATUS field
    % ------------------------------------------------------------

    rawBase + "gyro" + gyroAxes + "--status--status";


    % ------------------------------------------------------------
    % Gyro readout BIT status
    % ------------------------------------------------------------

    rawBase + "gyro" + gyroAxes + ...
        "--status--readout_bit_status";


    % ------------------------------------------------------------
    % Gyro current BIT status
    % ------------------------------------------------------------

    rawBase + "gyro" + gyroAxes + ...
        "--status--current_bit_status";


    % ------------------------------------------------------------
    % Gyro dither BIT status
    % ------------------------------------------------------------

    rawBase + "gyro" + gyroAxes + ...
        "--status--dither_bit_status";


    % ------------------------------------------------------------
    % Gyro PLC BIT status
    % ------------------------------------------------------------

    rawBase + "gyro" + gyroAxes + ...
        "--status--plc_bit_status";


    % ------------------------------------------------------------
    % Gyro Status ID Counter
    % ------------------------------------------------------------

    rawBase + "gyro" + gyroAxes + ...
        "--status--status_id_counter";


    % ------------------------------------------------------------
    % Gyro theta LSB
    % ------------------------------------------------------------

    rawBase + "gyro" + gyroAxes + "--lsb";


    % ------------------------------------------------------------
    % Gyro theta MSB
    % ------------------------------------------------------------

    rawBase + "gyro" + gyroAxes + "--msb";


    % ------------------------------------------------------------
    % Gyro checksum
    % ------------------------------------------------------------

    rawBase + "gyro" + gyroAxes + "--checksum";
];


%% ================================================================
%  EXPECTED INTERNAL STRUCT NAMES
%  ================================================================

structBase = ...
    "HG9900_Raw_Data_Frame_Raptor_Counts_";

gyroIndices = ["1"; "2"; "3"];


expectedStructNames = [

    % ------------------------------------------------------------
    % Common fields
    % ------------------------------------------------------------

    "HG9900_Raw_Data_Frame_Msg_Number";

    "HG9900_Raw_Data_Frame_Timestamp";

    structBase + "accel_mux_tag";

    structBase + "accel_mux_word";


    % ------------------------------------------------------------
    % Gyro mux tag
    % ------------------------------------------------------------

    structBase + "gyro" + gyroIndices + "_mux_tag";


    % ------------------------------------------------------------
    % Gyro mux word
    % ------------------------------------------------------------

    structBase + "gyro" + gyroIndices + "_mux_word";


    % ------------------------------------------------------------
    % Gyro STATUS field
    % ------------------------------------------------------------

    structBase + "gyro" + gyroIndices + ...
        "_status_status";


    % ------------------------------------------------------------
    % Gyro readout BIT status
    % ------------------------------------------------------------

    structBase + "gyro" + gyroIndices + ...
        "_status_readout_bit_status";


    % ------------------------------------------------------------
    % Gyro current BIT status
    % ------------------------------------------------------------

    structBase + "gyro" + gyroIndices + ...
        "_status_current_bit_status";


    % ------------------------------------------------------------
    % Gyro dither BIT status
    % ------------------------------------------------------------

    structBase + "gyro" + gyroIndices + ...
        "_status_dither_bit_status";


    % ------------------------------------------------------------
    % Gyro PLC BIT status
    % ------------------------------------------------------------

    structBase + "gyro" + gyroIndices + ...
        "_status_plc_bit_status";


    % ------------------------------------------------------------
    % Gyro Status ID Counter
    % ------------------------------------------------------------

    structBase + "gyro" + gyroIndices + ...
        "_status_id_counter";


    % ------------------------------------------------------------
    % Gyro theta LSB
    % ------------------------------------------------------------

    structBase + "gyro" + gyroIndices + "_lsb";


    % ------------------------------------------------------------
    % Gyro theta MSB
    % ------------------------------------------------------------

    structBase + "gyro" + gyroIndices + "_msb";


    % ------------------------------------------------------------
    % Gyro checksum
    % ------------------------------------------------------------

    structBase + "gyro" + gyroIndices + "_checksum";
];


%% ================================================================
%  BASIC SIZE CHECKS
%  ================================================================

assert( ...
    numel(actualParamNames) == numel(actualStructNames), ...
    ['Parameter list and struct-name list have different sizes. ' ...
     'Parameters: %d, Struct names: %d'], ...
    numel(actualParamNames), ...
    numel(actualStructNames));


assert( ...
    numel(actualParamNames) == numel(expectedParamNames), ...
    ['Unexpected number of HG9900 parameters. ' ...
     'Expected: %d, Actual: %d'], ...
    numel(expectedParamNames), ...
    numel(actualParamNames));


%% ================================================================
%  COMPARE EACH MAPPING
%  ================================================================

paramMatch = actualParamNames == expectedParamNames;
structMatch = actualStructNames == expectedStructNames;

mappingMatch = paramMatch & structMatch;


%% Display complete mapping

results = table( ...
    expectedParamNames, ...
    actualParamNames, ...
    expectedStructNames, ...
    actualStructNames, ...
    mappingMatch, ...
    'VariableNames', { ...
        'ExpectedParameter', ...
        'ActualParameter', ...
        'ExpectedStruct', ...
        'ActualStruct', ...
        'Pass'});

disp(results);


%% ================================================================
%  REPORT FAILURES
%  ================================================================

if ~all(mappingMatch)

    fprintf('\n');
    fprintf('============================================================\n');
    fprintf('FAILED HG9900 MAPPINGS\n');
    fprintf('============================================================\n');

    failedRows = results(~mappingMatch, :);

    disp(failedRows);

    error( ...
        'HG9900 parameter mapping test failed for %d mapping(s).', ...
        sum(~mappingMatch));
end


%% ================================================================
%  SUCCESS
%  ================================================================

fprintf('\n');
fprintf('============================================================\n');
fprintf('STEP 1 HG9900 PARAMETER MAPPING PASSED\n');
fprintf('============================================================\n');
fprintf('Verified %d mappings.\n', numel(mappingMatch));
fprintf('\n');

fprintf('Gyro axis mapping:\n');
fprintf('  Gyro U -> gyro1\n');
fprintf('  Gyro V -> gyro2\n');
fprintf('  Gyro W -> gyro3\n');
fprintf('\n');

end
