"""
Generate constant-velocity position and velocity lookup tables for Simulink.

The generated MAT-file contains:

    trajectory_time
    position_x
    position_y
    position_z
    velocity_x
    velocity_y
    velocity_z

Position is calculated using:

    position(t) = initial_position + constant_velocity * t
"""

from pathlib import Path

import numpy as np
from scipy.io import savemat


def generate_constant_velocity_trajectory(
    initial_position: tuple[float, float, float],
    constant_velocity: tuple[float, float, float],
    duration: float,
    sample_rate: float,
    output_file: str = "gps_trajectory.mat",
) -> None:
    """
    Generate a constant-velocity trajectory and save it as a MAT-file.

    Parameters
    ----------
    initial_position
        Initial [x, y, z] position.

        Examples:
            NED:  [north, east, down] in meters
            ECEF: [x, y, z] in meters

    constant_velocity
        Constant [vx, vy, vz] velocity.

        Examples:
            NED:  [v_north, v_east, v_down] in m/s
            ECEF: [vx, vy, vz] in m/s

    duration
        Total trajectory duration in seconds.

    sample_rate
        Number of trajectory points per second in Hz.

    output_file
        Name of the generated MAT-file.
    """
    if duration <= 0.0:
        raise ValueError("duration must be greater than zero.")

    if sample_rate <= 0.0:
        raise ValueError("sample_rate must be greater than zero.")

    initial_position_array = np.asarray(
        initial_position,
        dtype=np.float64,
    )

    constant_velocity_array = np.asarray(
        constant_velocity,
        dtype=np.float64,
    )

    if initial_position_array.shape != (3,):
        raise ValueError("initial_position must contain exactly 3 values.")

    if constant_velocity_array.shape != (3,):
        raise ValueError("constant_velocity must contain exactly 3 values.")

    sample_time = 1.0 / sample_rate

    # Adding half a sample avoids accidentally omitting the final point
    # because of floating-point rounding.
    trajectory_time = np.arange(
        0.0,
        duration + 0.5 * sample_time,
        sample_time,
        dtype=np.float64,
    )

    # Each row is:
    # position[i, :] = initial_position + velocity * trajectory_time[i]
    position = (
        initial_position_array[None, :]
        + trajectory_time[:, None] * constant_velocity_array[None, :]
    )

    # Constant velocity at every trajectory point.
    velocity = np.tile(
        constant_velocity_array,
        (trajectory_time.size, 1),
    )

    mat_data = {
        # Column vector forms are convenient for Simulink tables.
        "trajectory_time": trajectory_time.reshape(-1, 1),

        "position_x": position[:, 0].reshape(-1, 1),
        "position_y": position[:, 1].reshape(-1, 1),
        "position_z": position[:, 2].reshape(-1, 1),

        "velocity_x": velocity[:, 0].reshape(-1, 1),
        "velocity_y": velocity[:, 1].reshape(-1, 1),
        "velocity_z": velocity[:, 2].reshape(-1, 1),

        # Optional metadata.
        "trajectory_sample_time": np.array(
            [[sample_time]],
            dtype=np.float64,
        ),
        "trajectory_sample_rate": np.array(
            [[sample_rate]],
            dtype=np.float64,
        ),
        "trajectory_duration": np.array(
            [[duration]],
            dtype=np.float64,
        ),
    }

    output_path = Path(output_file)
    savemat(output_path, mat_data)

    print(f"Saved trajectory to: {output_path.resolve()}")
    print(f"Number of points: {trajectory_time.size}")
    print(f"Sample time: {sample_time:.9f} seconds")
    print(f"Initial position: {initial_position_array}")
    print(f"Constant velocity: {constant_velocity_array}")
    print(f"Final position: {position[-1, :]}")


if __name__ == "__main__":
    generate_constant_velocity_trajectory(
        initial_position=(
            1000.0,  # x0
            2000.0,  # y0
            -500.0,  # z0
        ),
        constant_velocity=(
            25.0,  # vx
            5.0,   # vy
            0.0,   # vz
        ),
        duration=60.0,
        sample_rate=20.0,
        output_file="gps_trajectory.mat",
    )
