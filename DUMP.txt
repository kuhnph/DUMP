function [varBus] = testCases(t,k)
    X0_ECEF = [-8766956.82787788;-14810913.4476751;11862335.4911620];
    varBus.A_ECEF  = zeros(3,1);
    varBus.V_ECEF  = zeros(3,1);
    varBus.X_ECEF  = X0_ECEF;

    varBus.q_be    = zeros(4,1);
    varBus.q_bf    = zeros(4,1);
    varBus.q_bi    = zeros(4,1);

    varBus.Euler321 = zeros(3,1);

    varBus.w_be    = zeros(3,1);
    varBus.w_bf    = zeros(3,1);
    varBus.w_bi    = zeros(3,1);

    varBus.wDot_bi = zeros(3,1);
    switch k
        case 1
            %Maintain bus of 0s
        
        case 2
            %Linear Motion
            varBus.V_ECEF = [-8634.21920996787; -12.7701137038839; 1280.68497382080];
            varBus.X_ECEF = X0_ECEF + varBus.V_ECEF * t;

    case 3
        % Body angular velocity
        w_bf = [0.1;
                0.2;
                0.3];

        varBus.w_bf = w_bf;
        omega = norm(w_bf);
        
        if omega > 0
            theta = omega*t;
            u = w_bf / omega;
            varBus.q_bf = [cos(theta/2);    %Scalar first quaternion?
                        u*sin(theta/2)];
        else
            % no rotation
            varBus.q_bf = [1;0;0;0];

        end

    end


end

         'wDot_bi'};

        elems(i) = Simulink.BusElement;
        elems(i).Name = names{i};
        elems(i).Dimensions = [4 1];
        elems(i).DataType = 'double';
    end

end

StateBus = Simulink.Bus;
StateBus.Elements = elems;

assignin('base','StateBus',StateBus);
