static int getRTDBDoubleArray(
    const char *name,
    double *output,
    int outputSize)
{
    ItemValueDef value = {};

    // Tell RTDB that we already allocated storage.
    value.value.Val.array_d = output;
    value.value.numMallocedElements = outputSize;

    int status = ccurRTDB_getCVTValue(name, &value);

    if (status != 0)
    {
        printf("Failed to read RTDB variable: %s, status=%d\n",
               name, status);
        return status;
    }

    if (value.value.numElements != outputSize)
    {
        printf(
            "Unexpected array size for %s: expected %d, got %d\n",
            name,
            outputSize,
            value.value.numElements
        );

        return -1;
    }

    return 0;
}
