%% Store gyro status fields directly

% Gyro status
gyroTagName = 'gyro_status';
mux.gyro(g).(gyroTagName).AbsoluteReceivedTime = absRxTimeMux;
mux.gyro(g).(gyroTagName).time = timestampMux;
mux.gyro(g).(gyroTagName).value = uint8(gyroStatusStatus);
mux.gyro(g).(gyroTagName).id = NaN;
mux.gyro(g).(gyroTagName).unit = 'counts';
mux.gyro(g).(gyroTagName).comment = 'Gyro status byte.';
