#!/usr/bin/env python3

import sys
import os
import csv
import bisect
import gc

import simwbPath
import simwbRT


# =============================================================================
# RTDB VARIABLE NAMES
# =============================================================================

# Simulation time supplied through the SimWB RTDB
TIME_RTDB = "SW_T_Simulation"

# Trajectory outputs consumed by the Spirent model
X_ECEF_RTDB = "SW_Xecef"
V_ECEF_RTDB = "SW_Vecef"
A_ECEF_RTDB = "SW_Aecef"
J_ECEF_RTDB = "SW_Jecef"
EULER_RTDB  = "SW_Euler"


# =============================================================================
# TRAJECTORY FILE
# =============================================================================

# Place firefly_trac.csv in the same directory as this Python model.
MODEL_DIRECTORY = os.path.dirname(os.path.abspath(__file__))
CSV_FILE = os.path.join(MODEL_DIRECTORY, "firefly_trac.csv")


# =============================================================================
# CSV FORMAT
# =============================================================================
#
# Expected column order:
#
#  0   time
#
#  1   X_ECEF_1
#  2   X_ECEF_2
#  3   X_ECEF_3
#
#  4   V_ECEF_1
#  5   V_ECEF_2
#  6   V_ECEF_3
#
#  7   A_ECEF_1
#  8   A_ECEF_2
#  9   A_ECEF_3
#
# 10   J_ECEF_1
# 11   J_ECEF_2
# 12   J_ECEF_3
#
# 13   yaw
# 14   pitch
# 15   roll
#
# =============================================================================


class Trajectory:

    def __init__(self, filename):

        self.time = []
        self.data = []

        self._load_csv(filename)

        if len(self.time) < 2:
            raise RuntimeError(
                "Trajectory file must contain at least two valid data rows."
            )

        # Make sure time increases monotonically
        for i in range(1, len(self.time)):
            if self.time[i] <= self.time[i - 1]:
                raise RuntimeError(
                    "Trajectory time is not strictly increasing at index {}."
                    .format(i)
                )


    def _load_csv(self, filename):

        with open(filename, "r") as f:

            reader = csv.reader(f)

            for row in reader:

                # Each valid row must contain 16 values
                if len(row) != 16:
                    continue

                try:
                    values = [
                        float(value.strip())
                        for value in row
                    ]

                except ValueError:
                    # Allows the CSV to contain a text header
                    continue


                # Column 0 = time
                self.time.append(values[0])

                # Columns 1:16 = trajectory data
                self.data.append(values[1:])


    def sample(self, t):
        """
        Interpolate the trajectory at simulation time t.

        Returns:

            values[0:3]    X_ECEF
            values[3:6]    V_ECEF
            values[6:9]    A_ECEF
            values[9:12]   J_ECEF
            values[12:15]  yaw, pitch, roll
        """

        # Before the beginning of the trajectory,
        # hold the first trajectory point.
        if t <= self.time[0]:
            return self.data[0]


        # After the end of the trajectory,
        # hold the final trajectory point.
        if t >= self.time[-1]:
            return self.data[-1]


        # Find index i such that:
        #
        # self.time[i] <= t < self.time[i + 1]
        #
        i = bisect.bisect_right(self.time, t) - 1


        t0 = self.time[i]
        t1 = self.time[i + 1]

        y0 = self.data[i]
        y1 = self.data[i + 1]


        # Linear interpolation fraction
        alpha = (t - t0) / (t1 - t0)


        # Interpolate all 15 trajectory values
        values = [
            y0[j] + alpha * (y1[j] - y0[j])
            for j in range(15)
        ]

        return values


# =============================================================================
# RTDB HELPER
# =============================================================================

def get_rtdb_index(name):

    index = simwbRT.getItemIndex(name)

    if index < 0:
        raise RuntimeError(
            "Could not find RTDB variable: {}".format(name)
        )

    return index


# =============================================================================
# MAIN SIMWB MODEL
# =============================================================================

def main():

    # -------------------------------------------------------------------------
    # Map the active SimWB RTDB
    # -------------------------------------------------------------------------

    if simwbRT.mapRTDB() < 0:
        raise RuntimeError(
            "Failed to map active SimWB RTDB."
        )


    # -------------------------------------------------------------------------
    # Scheduler / log name
    # -------------------------------------------------------------------------

    if len(sys.argv) > 1:

        scheduler_name = sys.argv[1]
        log_name = os.path.basename(sys.argv[1])

    else:

        # Useful if manually executing during debugging
        scheduler_name = "TrajectoryModel"
        log_name = "TrajectoryModel"


    simwbRT.initLog(log_name)


    # -------------------------------------------------------------------------
    # Load trajectory ONCE during initialization
    # -------------------------------------------------------------------------

    try:

        trajectory = Trajectory(CSV_FILE)

    except Exception as error:

        simwbRT.logMessage(
            "Failed to load trajectory: {}".format(error)
        )

        simwbRT.initFailed()
        return


    # -------------------------------------------------------------------------
    # Find RTDB variables ONCE during initialization
    # -------------------------------------------------------------------------

    try:

        time_index = get_rtdb_index(
            TIME_RTDB
        )

        x_ecef_index = get_rtdb_index(
            X_ECEF_RTDB
        )

        v_ecef_index = get_rtdb_index(
            V_ECEF_RTDB
        )

        a_ecef_index = get_rtdb_index(
            A_ECEF_RTDB
        )

        j_ecef_index = get_rtdb_index(
            J_ECEF_RTDB
        )

        euler_index = get_rtdb_index(
            EULER_RTDB
        )

    except Exception as error:

        simwbRT.logMessage(
            "RTDB initialization failed: {}".format(error)
        )

        simwbRT.initFailed()
        return


    # -------------------------------------------------------------------------
    # Preallocate output arrays
    # -------------------------------------------------------------------------

    x_ecef = [0.0, 0.0, 0.0]
    v_ecef = [0.0, 0.0, 0.0]
    a_ecef = [0.0, 0.0, 0.0]
    j_ecef = [0.0, 0.0, 0.0]

    euler = [0.0, 0.0, 0.0]


    # -------------------------------------------------------------------------
    # Initialization log message
    # -------------------------------------------------------------------------

    simwbRT.logMessage(
        "Trajectory loaded successfully: {} samples, "
        "time {:.6f} to {:.6f} seconds."
        .format(
            len(trajectory.time),
            trajectory.time[0],
            trajectory.time[-1]
        )
    )


    # -------------------------------------------------------------------------
    # Disable Python cyclic garbage collection before entering real-time loop
    # -------------------------------------------------------------------------

    gc.disable()


    # -------------------------------------------------------------------------
    # Register with SimWB scheduler
    # -------------------------------------------------------------------------

    if simwbRT.initSched(scheduler_name) < 0:

        simwbRT.logMessage(
            "Failed to initialize SimWB scheduler."
        )

        simwbRT.initFailed()
        return


    # =========================================================================
    # REAL-TIME LOOP
    # =========================================================================

    try:

        while True:

            # -----------------------------------------------------------------
            # Wait for next SimWB simulation frame
            # -----------------------------------------------------------------

            simwbRT.waitNextFrame()


            # -----------------------------------------------------------------
            # Read current SimWB simulation time
            # -----------------------------------------------------------------

            time_value = simwbRT.getCVTValueFromIndex(
                time_index
            )


            if time_value is None:
                continue


            t = float(
                time_value[0]
            )


            # -----------------------------------------------------------------
            # Interpolate trajectory at this simulation time
            # -----------------------------------------------------------------

            values = trajectory.sample(t)


            # -----------------------------------------------------------------
            # Position - ECEF
            # -----------------------------------------------------------------

            x_ecef[0] = values[0]
            x_ecef[1] = values[1]
            x_ecef[2] = values[2]


            # -----------------------------------------------------------------
            # Velocity - ECEF
            # -----------------------------------------------------------------

            v_ecef[0] = values[3]
            v_ecef[1] = values[4]
            v_ecef[2] = values[5]


            # -----------------------------------------------------------------
            # Acceleration - ECEF
            # -----------------------------------------------------------------

            a_ecef[0] = values[6]
            a_ecef[1] = values[7]
            a_ecef[2] = values[8]


            # -----------------------------------------------------------------
            # Jerk - ECEF
            # -----------------------------------------------------------------

            j_ecef[0] = values[9]
            j_ecef[1] = values[10]
            j_ecef[2] = values[11]


            # -----------------------------------------------------------------
            # Euler angles
            #
            # Existing Spirent RTDB variable:
            #
            #     SW_Euler = double[3]
            #
            # Mapping:
            #
            #     [0] = yaw
            #     [1] = pitch
            #     [2] = roll
            # -----------------------------------------------------------------

            euler[0] = values[12]
            euler[1] = values[13]
            euler[2] = values[14]


            # -----------------------------------------------------------------
            # Write trajectory into the shared RTDB
            #
            # These are the SAME RTDB variables consumed by the existing
            # Spirent Simulink model.
            # -----------------------------------------------------------------

            simwbRT.setCVTValueFromIndex(
                x_ecef_index,
                x_ecef
            )

            simwbRT.setCVTValueFromIndex(
                v_ecef_index,
                v_ecef
            )

            simwbRT.setCVTValueFromIndex(
                a_ecef_index,
                a_ecef
            )

            simwbRT.setCVTValueFromIndex(
                j_ecef_index,
                j_ecef
            )

            simwbRT.setCVTValueFromIndex(
                euler_index,
                euler
            )


    except KeyboardInterrupt:

        # SimWB normally stops the process when the test ends.
        pass


# =============================================================================
# ENTRY POINT
# =============================================================================

if __name__ == "__main__":
    main()
