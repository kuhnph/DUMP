import numpy as np
import pandas as pd
from scipy.io import loadmat

mat = loadmat("rocketLab_trajectory.mat", simplify_cells=True)

# "dat" is the MATLAB structure variable saved with:
# save("rocketLab_trajectory.mat", "dat")
dat = mat["dat"]

columns = {}

for name, values in dat.items():
    values = np.asarray(values)

    # MATLAB N×1 vector
    if values.ndim == 1:
        columns[name] = values

    # MATLAB N×M matrix
    elif values.ndim == 2:
        for j in range(values.shape[1]):
            columns[f"{name}_{j + 1}"] = values[:, j]

df = pd.DataFrame(columns)

display(df.head())
print(df.shape)
