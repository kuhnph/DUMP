void sendInitialMOT(
    Netio& netio,
    const double xEcef[3],
    const double euler[3],
    int vehicleNum,
    int motionModelNum,
    char* serverIP,
    int serverPort
)
{
    char motMSG[512];

    int lenMotMSG = snprintf(
        motMSG,
        sizeof(motMSG),

        "0,MOT,v%d_m%d,"
        "%.10f,%.10f,%.10f,"
        "0,0,0,0,0,0,0,"
        "%.10f,%.10f,%.10f",

        vehicleNum,
        motionModelNum,

        xEcef[0],
        xEcef[1],
        xEcef[2],

        euler[0],
        euler[1],
        euler[2]
    );


    // Make sure snprintf succeeded and did not overflow the buffer
    if(lenMotMSG < 0 || lenMotMSG >= (int)sizeof(motMSG))
    {
        ccurLog_printf(
            defaultLog,
            LOG_ERROR,
            "Failed to create initial MOT message"
        );

        return;
    }


    // Useful while debugging
    printCommandBytes(
        motMSG,
        (size_t)lenMotMSG
    );


    netio.sendRaw(
        motMSG,
        (size_t)lenMotMSG,
        serverIP,
        serverPort
    );


    ccurLog_printf(
        defaultLog,
        LOG_INFO,
        "sent initial MOT command: %s",
        motMSG
    );
}
