function plot_rsim_rapid(vars, RsimOut, rapid_Output)
% vars: cell array or string array of variable names
% Example:
% vars = {'Lat','Lon','Alt'};
% plot_rsim_rapid(vars, RsimOut, rapid_Output)

    if ischar(vars)
        vars = {vars};
    elseif isstring(vars)
        vars = cellstr(vars);
    end

    for k = 1:numel(vars)
        varName = vars{k};

        % ---- RSIM plot ----
        if isfield(RsimOut, 'TALO') && isfield(RsimOut, varName)
            figure;
            plot(RsimOut.TALO, RsimOut.(varName), 'LineWidth', 1.5);
            xlabel('TALO');
            ylabel(varName, 'Interpreter', 'none');
            title('RSIM Out', 'Interpreter', 'none');
            grid on;
        else
            warning('RsimOut is missing TALO or %s', varName);
        end

        % ---- Rapid Accel plot ----
        if isfield(rapid_Output, 'TALO') && isfield(rapid_Output, varName)
            figure;
            plot(rapid_Output.TALO, rapid_Output.(varName), 'LineWidth', 1.5);
            xlabel('TALO');
            ylabel(varName, 'Interpreter', 'none');
            title('Rapid Accel Out', 'Interpreter', 'none');
            grid on;
        else
            warning('rapid_Output is missing TALO or %s', varName);
        end
    end
end
