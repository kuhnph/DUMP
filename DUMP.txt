#!/usr/bin/env python2

import sys
import os
import csv
import bisect

import simwbPath
import simwbRT
import simwbConstants


# =============================================================================
# RTDB NAMES
# =============================================================================

TIME_RTDB   = "SW_T_Simulation"

X_ECEF_RTDB = "SW_Xecef"
V_ECEF_RTDB = "SW_Vecef"
A_ECEF_RTDB = "SW_Aecef"
J_ECEF_RTDB = "SW_Jecef"
EULER_RTDB  = "SW_Euler"


# =============================================================================
# TRAJECTORY FILE
# =============================================================================

MODEL_DIRECTORY = os.path.dirname(os.path.abspath(__file__))
CSV_FILE = os.path.join(MODEL_DIRECTORY, "firefly_trac.csv")


# =============================================================================
# TRAJECTORY CLASS
# =============================================================================

class Trajectory(object):

    def __init__(self, filename):

        self.time = []
        self.data = []

        self.load_csv(filename)

        if len(self.time) < 2:
            raise RuntimeError(
                "Trajectory file must contain at least two valid rows."
            )

        for i in range(1, len(self.time)):
            if self.time[i] <= self.time[i - 1]:
                raise RuntimeError(
                    "Trajectory time must be strictly increasing."
                )


    def load_csv(self, filename):

        f = open(filename, "r")

        try:

            reader = csv.reader(f)

            for row in reader:

                if len(row) != 16:
                    continue

                try:
                    values = [float(value.strip()) for value in row]

                except ValueError:
                    # Allows a header row
                    continue


                self.time.append(values[0])

                # Store the 15 trajectory values
                self.data.append(values[1:])

        finally:

            f.close()


    def sample(self, t):

        # Hold first point before trajectory starts
        if t <= self.time[0]:
            return self.data[0]

        # Hold last point after trajectory ends
        if t >= self.time[-1]:
            return self.data[-1]


        # Find lower trajectory sample
        i = bisect.bisect_right(self.time, t) - 1

        t0 = self.time[i]
        t1 = self.time[i + 1]

        y0 = self.data[i]
        y1 = self.data[i + 1]


        alpha = (t - t0) / (t1 - t0)


        values = []

        for j in range(15):

            value = y0[j] + alpha * (y1[j] - y0[j])

            values.append(value)


        return values


# =============================================================================
# CONNECT TO SIMWB
# =============================================================================

simwbRT.mapRTDB()

simwbRT.initLog(
    os.path.basename(sys.argv[1])
)

simwbRT.initSched(
    sys.argv[1]
)


# =============================================================================
# LOAD TRAJECTORY
# =============================================================================

try:

    trajectory = Trajectory(CSV_FILE)

    simwbRT.logMessage(
        "Loaded trajectory: %d samples, %.6f to %.6f seconds"
        % (
            len(trajectory.time),
            trajectory.time[0],
            trajectory.time[-1]
        )
    )

except Exception as error:

    simwbRT.logMessage(
        "Trajectory load failed: %s"
        % str(error)
    )

    simwbRT.initFailed()

    sys.exit(1)


# =============================================================================
# SCHEDULER INITIALIZATION CYCLES
# =============================================================================

try:

    for i in range(simwbConstants.SCHED_INITCYCLES):

        simwbRT.waitNextFrame()


    # =========================================================================
    # MAIN REAL-TIME LOOP
    # =========================================================================

    while True:

        # Wait until SimWB releases this process for the next frame
        simwbRT.waitNextFrame()


        # ---------------------------------------------------------------------
        # Read simulation time from the RTDB
        # ---------------------------------------------------------------------

        time_value = simwbRT.getCVTValue(
            TIME_RTDB
        )

        t = float(
            time_value[0]
        )


        # ---------------------------------------------------------------------
        # Interpolate CSV trajectory
        # ---------------------------------------------------------------------

        values = trajectory.sample(t)


        # ---------------------------------------------------------------------
        # Map trajectory values
        # ---------------------------------------------------------------------

        X_ECEF = [
            values[0],
            values[1],
            values[2]
        ]

        V_ECEF = [
            values[3],
            values[4],
            values[5]
        ]

        A_ECEF = [
            values[6],
            values[7],
            values[8]
        ]

        J_ECEF = [
            values[9],
            values[10],
            values[11]
        ]

        Euler = [
            values[12],
            values[13],
            values[14]
        ]


        # ---------------------------------------------------------------------
        # Write directly into shared RTDB variables
        # ---------------------------------------------------------------------

        simwbRT.setCVTValue(
            X_ECEF_RTDB,
            X_ECEF
        )

        simwbRT.setCVTValue(
            V_ECEF_RTDB,
            V_ECEF
        )

        simwbRT.setCVTValue(
            A_ECEF_RTDB,
            A_ECEF
        )

        simwbRT.setCVTValue(
            J_ECEF_RTDB,
            J_ECEF
        )

        simwbRT.setCVTValue(
            EULER_RTDB,
            Euler
        )


except KeyboardInterrupt:

    simwbRT.logMessage(
        "Trajectory model shutting down."
    )

    sys.exit(0)
