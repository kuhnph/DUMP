%% Gyro parameter names

gyroAxes = ["U", "V", "W"];
gyroIndices = 1:options.GyroCount;


%% Gyro mux tag
% External:
%   Gyro_U--Mux_Tag
%   Gyro_V--Mux_Tag
%   Gyro_W--Mux_Tag
%
% Internal:
%   gyro1_mux_tag
%   gyro2_mux_tag
%   gyro3_mux_tag

gyroTagNames = compose( ...
    strcat(options.ParamPrefix, options.ParamGyroTagTemplate), ...
    gyroAxes);

gyroStructTagNames = compose( ...
    'HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_mux_tag', ...
    gyroIndices);


%% Gyro mux word / byte
% External:
%   Gyro_U--Mux_Byte
%   Gyro_V--Mux_Byte
%   Gyro_W--Mux_Byte
%
% Internal names remain mux_word so the existing decoder interface
% does not need to change unnecessarily.

gyroWordNames = compose( ...
    strcat(options.ParamPrefix, options.ParamGyroWordTemplate), ...
    gyroAxes);

gyroStructWordNames = compose( ...
    'HG9900_Raw_Data_Frame_Raptor_Counts_gyro%d_mux_word', ...
    gyroIndices);
