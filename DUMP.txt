#!/usr/bin/env python2

import sys
import os

import simwbPath
import simwbRT
import simwbConstants


TIME_RTDB = "SW_T_Simulation"


simwbRT.mapRTDB()

simwbRT.initLog(
    os.path.basename(sys.argv[1])
)

simwbRT.initSched(
    sys.argv[1]
)


frame_rate = float(
    simwbRT.getFrameRate()
)

dt = 1.0 / frame_rate


try:

    for i in range(simwbConstants.SCHED_INITCYCLES):
        simwbRT.waitNextFrame()


    start_frame = simwbRT.getFrameCount()


    while True:

        simwbRT.waitNextFrame()

        frame_count = simwbRT.getFrameCount()

        relative_frame = frame_count - start_frame

        t_simulation = float(relative_frame) * dt


        # IMPORTANT:
        # SW_T_Simulation is a scalar double,
        # so write a scalar rather than [scalar].
        simwbRT.setCVTValue(
            TIME_RTDB,
            t_simulation
        )


        if relative_frame % 1200 == 0:

            # Read the value back immediately after writing it.
            readback = simwbRT.getCVTValue(
                TIME_RTDB
            )

            simwbRT.logMessage(
                "frame=%d calculated=%f RTDB readback=%s"
                % (
                    frame_count,
                    t_simulation,
                    str(readback)
                )
            )


except KeyboardInterrupt:

    simwbRT.logMessage(
        "pyTime shutting down"
    )

    sys.exit(0)
