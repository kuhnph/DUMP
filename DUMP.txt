import h5py

def print_item(name, obj):
    if isinstance(obj, h5py.Dataset):
        print(f"Dataset: {name}")
        print(f"    shape = {obj.shape}")
        print(f"    dtype = {obj.dtype}")
    elif isinstance(obj, h5py.Group):
        print(f"Group: {name}")

with h5py.File("XX.h5", "r") as f:
    f.visititems(print_item)
