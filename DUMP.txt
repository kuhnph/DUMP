// ============================================================================
// SPIRent STARTUP COMMAND SEQUENCE
//
// idx = 2 : Open scenario
// idx = 3 : Configure external trigger
// idx = 4 : Send initial MOT state
// idx = 5 : Arm scenario
// idx = 6 : Run scenario / enter trigger wait
// ============================================================================


// ----------------------------------------------------------------------------
// idx = 2
// Open scenario
// ----------------------------------------------------------------------------

if(idx == 2 && !scenarioOpened)
{
    printCommandBytes(
        openMSG,
        lenOpenMSG
    );

    netio.sendRaw(
        (char*)openMSG,
        lenOpenMSG,
        serverIP,
        serverPort
    );

    scenarioOpened = true;

    ccurLog_printf(
        defaultLog,
        LOG_INFO,
        "sent open command: %s",
        openMSG
    );
}


// ----------------------------------------------------------------------------
// idx = 3
// Configure external trigger
// ----------------------------------------------------------------------------

if(idx == 3 && scenarioOpened && !triggerSet)
{
    printCommandBytes(
        triggerMSG,
        lenTriggerMSG
    );

    netio.sendRaw(
        (char*)triggerMSG,
        lenTriggerMSG,
        serverIP,
        serverPort
    );

    triggerSet = true;

    ccurLog_printf(
        defaultLog,
        LOG_INFO,
        "sent trigger command: %s",
        triggerMSG
    );
}


// ----------------------------------------------------------------------------
// idx = 4
// Send initial motion state
//
// Expected format from the original Simulink compose block:
//
// 0,MOT,v<vehicle>_m<motion model>,
// X_ECEF_1,
// X_ECEF_2,
// X_ECEF_3,
// 0,0,0,0,0,0,0,
// yaw,
// pitch,
// roll
//
// For the first test, xEcef[] and euler[] can be hard-coded to the first
// trajectory sample.
// ----------------------------------------------------------------------------

if(idx == 4 && triggerSet && !initialMotSent)
{
    char motMSG[512];

    int lenMotMSG = snprintf(
        motMSG,
        sizeof(motMSG),

        "0,MOT,v%d_m%d,"
        "%.10f,%.10f,%.10f,"
        "0,0,0,0,0,0,0,"
        "%.10f,%.10f,%.10f",

        vehicleNum,
        motionModelNum,

        xEcef[0],
        xEcef[1],
        xEcef[2],

        euler[0],
        euler[1],
        euler[2]
    );


    // Make sure snprintf succeeded and did not overflow motMSG
    if(lenMotMSG < 0 || lenMotMSG >= (int)sizeof(motMSG))
    {
        ccurLog_printf(
            defaultLog,
            LOG_ERROR,
            "Failed to create initial MOT message"
        );
    }
    else
    {
        printCommandBytes(
            motMSG,
            (size_t)lenMotMSG
        );

        netio.sendRaw(
            motMSG,
            (size_t)lenMotMSG,
            serverIP,
            serverPort
        );

        initialMotSent = true;

        ccurLog_printf(
            defaultLog,
            LOG_INFO,
            "sent initial MOT command: %s",
            motMSG
        );
    }
}


// ----------------------------------------------------------------------------
// idx = 5
// Arm scenario
// ----------------------------------------------------------------------------

if(idx == 5 && initialMotSent && !scenarioArmed)
{
    printCommandBytes(
        armMSG,
        lenArmMSG
    );

    netio.sendRaw(
        (char*)armMSG,
        lenArmMSG,
        serverIP,
        serverPort
    );

    scenarioArmed = true;

    ccurLog_printf(
        defaultLog,
        LOG_INFO,
        "sent arm command: %s",
        armMSG
    );
}


// ----------------------------------------------------------------------------
// idx = 6
// Run scenario
//
// With TR,1 previously configured, this should request the transition into
// trigger wait.
// ----------------------------------------------------------------------------

if(idx == 6 && scenarioArmed && !scenarioRunning)
{
    printCommandBytes(
        runMSG,
        lenRunMSG
    );

    netio.sendRaw(
        (char*)runMSG,
        lenRunMSG,
        serverIP,
        serverPort
    );

    scenarioRunning = true;

    ccurLog_printf(
        defaultLog,
        LOG_INFO,
        "sent run command: %s",
        runMSG
    );
}
